## 1. 音频设计哲学

### 1.1 核心设计原则

```
🎵 Microverse音频设计五大原则:
1. 温馨舒适 - 营造Stardew Valley式的温暖氛围
2. 动态适应 - 音乐随时间、天气、场景变化
3. 非侵入性 - 音效不打扰,可长时间聆听
4. 涌现感 - 音频增强AI自主生活的真实感
5. 情感共鸣 - 通过音乐传递AI的情绪和故事
```

### 1.2 音频风格参考

```
【音乐风格】
主要参考:
- Stardew Valley (温馨、田园、爵士)
- Animal Crossing (轻快、日系、治愈)
- Celeste (电子、氛围、情感)

风格定位:
- 主旋律: 温暖的钢琴+吉他
- 节奏: 轻快但不急促(80-120 BPM)
- 编曲: 简约,避免过度复杂
- 情绪: 积极、希望、生活化

【音效风格】
- 像素化音效: 8-bit风格的清脆音效
- 环境音效: 真实但简化的自然音
- UI音效: 短促、清晰、友好
- 对话音效: 类似Animal Crossing的拟声
```

## 2. 音乐系统设计

### 2.1 动态音乐系统架构

```
【音乐层级结构】

Level 1: 基础轨道 (Base Tracks)
- 始终播放
- 定义主旋律和基础节奏
- 根据时间段切换

Level 2: 场景层 (Scene Layers)
- 根据AI所在场景叠加
- 办公室: 键盘声、打印机
- 咖啡馆: 对话声、咖啡机
- 室外: 风声、鸟鸣

Level 3: 情绪层 (Emotion Layers)
- 根据AI情绪动态混音
- 快乐: 增加铃铛、口哨
- 悲伤: 减少节奏,增加弦乐
- 紧张: 增加节奏,低频

Level 4: 事件层 (Event Layers)
- 特殊事件触发
- AI对话: 对话音效
- 节日: 节日音乐
- 重大事件: 特殊音效
```

### 2.3 情绪动态混音

#### 2.3.1 AI情绪音乐映射

```
【快乐 (Happy, Mood 70-100)】
音乐调整:
- 增加高频乐器(口哨、铃铛、木琴)
- 提升节奏(+5 BPM)
- 增加混响(营造开阔感)
- 音量稍微增加(+2dB)

【平静 (Calm, Mood 40-70)】
音乐调整:
- 保持基础轨道
- 适度的情绪层
- 标准混音

【悲伤 (Sad, Mood 0-40)】
音乐调整:
- 减少高频,增加低频
- 降低节奏(-10 BPM)
- 增加弦乐(低音提琴、大提琴)
- 减少打击乐
- 音量降低(-3dB)

【紧张 (Anxious, Stress 70-100)】
音乐调整:
- 增加低频脉冲
- 加快节奏(+10 BPM)
- 增加不协和音(轻微)
- 减少旋律,增加氛围
```

#### 2.3.2 情绪混音实现

```gdscript
# script/audio/EmotionMusicMixer.gd
class_name EmotionMusicMixer extends Node

# 情绪音乐层的AudioStreamPlayer
var happy_layer: AudioStreamPlayer
var sad_layer: AudioStreamPlayer
var anxious_layer: AudioStreamPlayer

# 当前情绪权重
var emotion_weights = {
	"happy": 0.0,
	"sad": 0.0,
	"anxious": 0.0,
	"calm": 1.0  # 默认平静
}

# 平滑过渡时间
const TRANSITION_TIME = 3.0  # 3秒渐变

func _ready():
	_setup_emotion_layers()

func _setup_emotion_layers():
	# 创建情绪层的AudioStreamPlayer
	happy_layer = AudioStreamPlayer.new()
	happy_layer.stream = load("res://assets/audio/music/layers/happy_layer.ogg")
	happy_layer.volume_db = -80  # 初始静音
	happy_layer.bus = "Music"
	add_child(happy_layer)
	happy_layer.play()

	sad_layer = AudioStreamPlayer.new()
	sad_layer.stream = load("res://assets/audio/music/layers/sad_layer.ogg")
	sad_layer.volume_db = -80
	sad_layer.bus = "Music"
	add_child(sad_layer)
	sad_layer.play()

	anxious_layer = AudioStreamPlayer.new()
	anxious_layer.stream = load("res://assets/audio/music/layers/anxious_layer.ogg")
	anxious_layer.volume_db = -80
	anxious_layer.bus = "Music"
	add_child(anxious_layer)
	anxious_layer.play()

	# 确保所有层同步
	_sync_layers()

# 根据AI情绪更新混音
func update_emotion_mix(ai_mood: int, ai_stress: int):
	# 计算情绪权重
	var new_weights = _calculate_emotion_weights(ai_mood, ai_stress)

	# 平滑过渡到新权重
	_transition_to_weights(new_weights)

func _calculate_emotion_weights(mood: int, stress: int) -> Dictionary:
	var weights = {
		"happy": 0.0,
		"sad": 0.0,
		"anxious": 0.0,
		"calm": 0.0
	}

	# 快乐权重 (mood高,stress低)
	if mood > 70 and stress < 50:
		weights["happy"] = min((mood - 70) / 30.0, 1.0)

	# 悲伤权重 (mood低)
	if mood < 40:
		weights["sad"] = min((40 - mood) / 40.0, 1.0)

	# 焦虑权重 (stress高)
	if stress > 70:
		weights["anxious"] = min((stress - 70) / 30.0, 1.0)

	# 平静权重 (默认)
	var total_emotion = weights["happy"] + weights["sad"] + weights["anxious"]
	weights["calm"] = max(0.0, 1.0 - total_emotion)

	return weights

# 平滑过渡到新权重
func _transition_to_weights(new_weights: Dictionary):
	# 使用Tween实现平滑音量过渡
	var tween = create_tween()
	tween.set_parallel(true)

	# Happy层
	var happy_target_db = _weight_to_db(new_weights["happy"])
	tween.tween_property(happy_layer, "volume_db", happy_target_db, TRANSITION_TIME)

	# Sad层
	var sad_target_db = _weight_to_db(new_weights["sad"])
	tween.tween_property(sad_layer, "volume_db", sad_target_db, TRANSITION_TIME)

	# Anxious层
	var anxious_target_db = _weight_to_db(new_weights["anxious"])
	tween.tween_property(anxious_layer, "volume_db", anxious_target_db, TRANSITION_TIME)

	# 更新权重记录
	emotion_weights = new_weights

# 权重转音量dB (0.0 = -80dB静音, 1.0 = 0dB全音量)
func _weight_to_db(weight: float) -> float:
	if weight <= 0.0:
		return -80.0
	return linear_to_db(weight)

# 同步所有层的播放位置
func _sync_layers():
	var base_position = happy_layer.get_playback_position()
	sad_layer.seek(base_position)
	anxious_layer.seek(base_position)
```

### 3.2 音效触发系统

```gdscript
# script/audio/SFXManager.gd
class_name SFXManager extends Node

# 音效池
var sfx_pools = {}

# AudioStreamPlayer池(用于同时播放多个音效)
var player_pool = []
const MAX_PLAYERS = 32

func _ready():
	_load_all_sfx()
	_create_player_pool()

# 加载所有音效到内存
func _load_all_sfx():
	sfx_pools["ui"] = _load_sfx_category("res://assets/audio/sfx/ui/")
	sfx_pools["character"] = _load_sfx_category("res://assets/audio/sfx/character/")
	sfx_pools["ambient"] = _load_sfx_category("res://assets/audio/sfx/ambient/")
	sfx_pools["scene"] = _load_sfx_category("res://assets/audio/sfx/scene/")

func _load_sfx_category(path: String) -> Dictionary:
	var sfx_dict = {}
	var dir = DirAccess.open(path)
	if dir:
		dir.list_dir_begin()
		var file_name = dir.get_next()
		while file_name != "":
			if file_name.ends_with(".wav") or file_name.ends_with(".ogg"):
				var sfx_name = file_name.get_basename()
				sfx_dict[sfx_name] = load(path + file_name)
			file_name = dir.get_next()
		dir.list_dir_end()
	return sfx_dict

# 创建AudioStreamPlayer池
func _create_player_pool():
	for i in range(MAX_PLAYERS):
		var player = AudioStreamPlayer.new()
		player.bus = "SFX"
		add_child(player)
		player_pool.append(player)

# 播放音效
func play_sfx(category: String, sfx_name: String, volume_db: float = 0.0, pitch_scale: float = 1.0):
	if not sfx_pools.has(category):
		push_error("[SFXManager] Category not found: %s" % category)
		return

	if not sfx_pools[category].has(sfx_name):
		push_warning("[SFXManager] SFX not found: %s in %s" % [sfx_name, category])
		return

	# 从池中获取空闲的AudioStreamPlayer
	var player = _get_available_player()
	if player == null:
		push_warning("[SFXManager] No available player (all %d players busy)" % MAX_PLAYERS)
		return

	# 设置并播放音效
	player.stream = sfx_pools[category][sfx_name]
	player.volume_db = volume_db
	player.pitch_scale = pitch_scale
	player.play()

# 播放随机变体音效 (例如鸟鸣有多个变体)
func play_sfx_random(category: String, sfx_prefix: String, volume_db: float = 0.0):
	var matching_sfx = []
	for sfx_name in sfx_pools[category].keys():
		if sfx_name.begins_with(sfx_prefix):
			matching_sfx.append(sfx_name)

	if matching_sfx.size() > 0:
		var random_sfx = matching_sfx[randi() % matching_sfx.size()]
		play_sfx(category, random_sfx, volume_db)

# 获取空闲的AudioStreamPlayer
func _get_available_player() -> AudioStreamPlayer:
	for player in player_pool:
		if not player.playing:
			return player
	return null

# 播放脚步声 (带地表判断)
func play_footstep(character_position: Vector2):
	# 检测角色所在地表类型
```

### 7.2 音频加载策略

```gdscript
# script/audio/AudioResourceManager.gd
class_name AudioResourceManager extends Node

# 音频资源池
var loaded_music = {}
var loaded_sfx = {}

# 加载策略
enum LoadStrategy {
	PRELOAD,  # 启动时预加载
	LAZY,     # 按需加载
	STREAM    # 流式加载(大文件)
}

# 音频资源配置
const AUDIO_CONFIG = {
	"music": {
		"strategy": LoadStrategy.STREAM,  # 音乐文件大,流式加载
		"path": "res://assets/audio/music/"
	},
	"sfx_ui": {
		"strategy": LoadStrategy.PRELOAD,  # UI音效小,预加载
		"path": "res://assets/audio/sfx/ui/"
	},
	"sfx_character": {
		"strategy": LoadStrategy.LAZY,  # 角色音效按需加载
		"path": "res://assets/audio/sfx/character/"
	},
	"sfx_ambient": {
		"strategy": LoadStrategy.LAZY,
		"path": "res://assets/audio/sfx/ambient/"
	}
}

func _ready():
	_load_preload_resources()

# 预加载必要资源
func _load_preload_resources():
	for category in AUDIO_CONFIG.keys():
		var config = AUDIO_CONFIG[category]
		if config.strategy == LoadStrategy.PRELOAD:
			_load_category(category, config.path)

func _load_category(category: String, path: String):
	var dir = DirAccess.open(path)
	if dir:
		dir.list_dir_begin()
		var file_name = dir.get_next()
		while file_name != "":
			if file_name.ends_with(".ogg") or file_name.ends_with(".wav"):
				var audio_stream = load(path + file_name)
				loaded_sfx[file_name.get_basename()] = audio_stream
			file_name = dir.get_next()
		dir.list_dir_end()
		print("[AudioResource] Loaded %s: %d files" % [category, loaded_sfx.size()])

# 获取音频资源 (按需加载)
func get_audio(category: String, audio_name: String) -> AudioStream:
	# 检查是否已加载
	if loaded_sfx.has(audio_name):
		return loaded_sfx[audio_name]

	# 按需加载
	var config = AUDIO_CONFIG.get(category, {})
	if config.is_empty():
		push_error("[AudioResource] Unknown category: %s" % category)
		return null

	var path = config.path + audio_name + ".ogg"
	if not FileAccess.file_exists(path):
		path = config.path + audio_name + ".wav"

	if FileAccess.file_exists(path):
		var audio_stream = load(path)
		loaded_sfx[audio_name] = audio_stream
		return audio_stream
	else:
		push_error("[AudioResource] Audio not found: %s" % path)
		return null
```

### 7.3 性能优化建议

```
【音频性能优化】

1. 限制同时播放音效数量:
   - 最多32个AudioStreamPlayer
   - 超出时跳过低优先级音效

2. 音效优先级:
   ✅ 高优先级: UI音效,对话拟声
   ✅ 中优先级: 角色脚步,场景音效
   ✅ 低优先级: 远处环境音

3. 音效距离衰减:
   - 远处音效降低音量
```

```gdscript
# script/audio/AudioDebugger.gd
# 音频调试工具 (开发模式)
extends Control

var audio_manager: Node

func _ready():

	debug_panel = Panel.new()
	debug_panel.set_anchors_preset(Control.PRESET_TOP_RIGHT)
	debug_panel.offset_left = -300
	debug_panel.offset_right = 0
	debug_panel.offset_top = 0
	debug_panel.offset_bottom = 400
	add_child(debug_panel)

	debug_label = Label.new()
	debug_label.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
	debug_panel.add_child(debug_label)

func _process(_delta):
	if not debug_label:
		return

	var debug_text = "[Audio Debug]\n\n"

	# 音量信息
	debug_text += "=== Volumes ===\n"
	debug_text += "Master: %.2f\n" % audio_manager.master_volume
	debug_text += "Music: %.2f\n" % audio_manager.music_volume
	debug_text += "SFX: %.2f\n" % audio_manager.sfx_volume
	debug_text += "\n"

	# 当前播放音乐
	debug_text += "=== Music ===\n"
	if audio_manager.music_manager:
		debug_text += "Current: %s\n" % audio_manager.music_manager.current_music
	debug_text += "\n"

	# 活跃音效数量
	debug_text += "=== SFX ===\n"
	if audio_manager.sfx_manager:
		var active_count = 0
		for player in audio_manager.sfx_manager.player_pool:
			if player.playing:
				active_count += 1
		debug_text += "Active Players: %d / %d\n" % [active_count, audio_manager.sfx_manager.MAX_PLAYERS]

	debug_label.text = debug_text

func _input(event):
	# F9切换调试面板
	if event is InputEventKey and event.keycode == KEY_F9 and event.pressed:
		debug_panel.visible = not debug_panel.visible
```
