### 核心性能指标

```yaml
帧率 (FPS):
  目标: 稳定60 FPS
  最低可接受: 45 FPS
  测试场景: 8个AI同时活动,2x速度

内存占用:
  启动时: < 200 MB
  运行1小时: < 400 MB
  运行10小时: < 800 MB
  运行100游戏日: < 1.5 GB
  峰值上限: < 2 GB

CPU占用:
  平均: < 20% (单核)
  峰值: < 40% (AI思考高峰期)
  空闲: < 5%

存档大小:
  新存档: < 1 MB
  运行30天: < 10 MB
  运行100天: < 25 MB
  理论上限: 50 MB

LLM调用:
  平均延迟: < 2秒
  队列长度: < 10个请求
  失败率: < 1%
  成本/小时: < $2.00

数据库查询:
  记忆查询: < 10ms
  事件查询: < 5ms
  关系查询: < 3ms
```

### 性能瓶颈预测

```yaml
潜在瓶颈:

1. 内存泄漏风险 (高):
   来源:
     - HTTPRequest未正确释放
     - 信号未断开连接
     - 事件数据无限累积
     - AI记忆无限增长

2. 数据累积 (高):
   来源:
     - 事件历史持续增长
     - 对话日志无限保存
     - 天气历史记录
     - 关系变化历史

3. LLM队列阻塞 (中):
   来源:
     - 多AI同时请求LLM
     - API响应慢
     - 请求失败未处理

4. UI渲染 (中):
   来源:
     - 大量对话气泡
     - 事件通知堆积
     - 实时数据更新

5. 存档大小膨胀 (中):
   来源:
     - 完整历史数据保存
     - 未压缩的JSON
     - 冗余数据存储
```

## 🗄️ 数据生命周期管理 (核心)

### 数据保留策略

**原则: 重要性 × 时效性 = 保留时长**

```yaml
数据分级保留策略:

Tier 1 - 核心数据 (永久保留):
  - AI基础信息 (性格、职业、年龄)
  - 当前状态 (位置、心情、精力、金钱)
  - 关系网络 (好感度、信任度、标签)
  - 重要记忆 (importance >= 80)
  - 里程碑事件 (成为恋人、冲突和解等)
  保留: 永久

Tier 2 - 重要数据 (30天):
  - 一般记忆 (importance 50-79)
  - 重要事件 (importance >= 60)
  - 任务完成记录
  - 工资发放记录
  - 冲突历史
  保留: 30游戏日
  淘汰策略: 30天后压缩为摘要

Tier 3 - 临时数据 (15天):
  - 不重要记忆 (importance < 50)
  - 日常事件 (importance < 60)
  - 普通对话记录
  - 天气变化历史
  - 位置移动历史
  保留: 15游戏日
  淘汰策略: 15天后删除

Tier 4 - 缓存数据 (7天):
  - LLM响应缓存
  - 路径计算结果
  - UI状态缓存
  - 临时计算数据
  保留: 7游戏日
  淘汰策略: 7天后删除

Tier 5 - 实时数据 (当前会话):
  - 当前对话上下文
  - 实时思考流
  - UI临时状态
  - 未完成的API请求
  保留: 会话期间
  淘汰策略: 游戏关闭时清除
```

### 数据清理器实现

```gdscript
# script/performance/DataLifecycleManager.gd
extends Node

# 单例
static var instance: DataLifecycleManager

# 清理配置
const CLEANUP_INTERVAL = 3600.0  # 每小时清理一次 (游戏时间)
const TIER_2_RETENTION_DAYS = 30
const TIER_3_RETENTION_DAYS = 15
const TIER_4_RETENTION_DAYS = 7

# 统计
var cleanup_stats = {
	"total_cleanups": 0,
	"memories_deleted": 0,
	"events_deleted": 0,
	"dialogues_deleted": 0,
	"total_space_freed": 0
}

var cleanup_timer: Timer

func _ready():
	instance = self

	# 设置定时清理
	cleanup_timer = Timer.new()
	cleanup_timer.wait_time = CLEANUP_INTERVAL
	cleanup_timer.one_shot = false
	cleanup_timer.timeout.connect(_on_cleanup_timer)
	add_child(cleanup_timer)
	cleanup_timer.start()

	print("[DataLifecycle] 数据生命周期管理器已启动")

# 定时清理
func _on_cleanup_timer():
	print("[DataLifecycle] 开始定时数据清理...")
	var start_time = Time.get_ticks_msec()

	perform_full_cleanup()

	var duration = Time.get_ticks_msec() - start_time
	print("[DataLifecycle] 清理完成,耗时 %d ms" % duration)

	cleanup_stats.total_cleanups += 1

# 执行完整清理
func perform_full_cleanup():
	# 1. 清理AI记忆
	cleanup_memories()

	# 2. 清理事件历史
	cleanup_events()

	# 3. 清理对话日志
	cleanup_dialogues()

	# 4. 清理天气历史
	cleanup_weather_history()

	# 5. 清理LLM缓存
	cleanup_llm_cache()

	# 6. 清理失效的对象引用
	cleanup_dead_references()

	# 7. 强制垃圾回收
	force_garbage_collection()

# ===== 记忆清理 =====

func cleanup_memories():
	print("[DataLifecycle] 清理AI记忆...")

	var total_deleted = 0
	var current_time = TimeSystem.get_timestamp()

	for ai in CharacterManager.get_all_characters():
		var ai_memories = MemoryManager.get_all_memories(ai.id)
		var to_delete = []

		for memory in ai_memories:
			var age_days = TimeSystem.calculate_duration_days(
				memory.timestamp,
				current_time
			)

			# Tier 3: 删除15天前的低重要性记忆
			if age_days > TIER_3_RETENTION_DAYS and memory.importance < 50:
				to_delete.append(memory.id)

			# Tier 2: 压缩30天前的中等重要性记忆
			elif age_days > TIER_2_RETENTION_DAYS and memory.importance < 80:
				_compress_memory(memory)

		# 执行删除
		for memory_id in to_delete:
			MemoryManager.delete_memory(ai.id, memory_id)
			total_deleted += 1

	cleanup_stats.memories_deleted += total_deleted
	print("[DataLifecycle] 删除了 %d 条记忆" % total_deleted)

# 压缩记忆 (保留摘要,删除详情)
func _compress_memory(memory: Dictionary):
	# 保留关键信息,删除详细内容
	if memory.content.length() > 100:
		memory.content = memory.content.substr(0, 100) + "..."
		memory.compressed = true

# ===== 事件清理 =====

func cleanup_events():
	print("[DataLifecycle] 清理事件历史...")

	var total_deleted = 0
	var current_time = TimeSystem.get_timestamp()

	# 获取已完成事件
	var completed_events = EventSystem.instance.completed_events

	var to_delete_indices = []

	for i in range(completed_events.size()):
		var event = completed_events[i]
		var age_days = TimeSystem.calculate_duration_days(
			event.ended_at,
			current_time
		)

		# 删除15天前的低重要性事件
		if age_days > TIER_3_RETENTION_DAYS and event.importance <= EventData.Importance.NORMAL:
			to_delete_indices.append(i)

		# 压缩30天前的重要事件
		elif age_days > TIER_2_RETENTION_DAYS and event.importance == EventData.Importance.MAJOR:
			_compress_event(event)

	# 从后往前删除,避免索引问题
	to_delete_indices.reverse()
	for idx in to_delete_indices:
		completed_events.remove_at(idx)
		total_deleted += 1

	cleanup_stats.events_deleted += total_deleted
	print("[DataLifecycle] 删除了 %d 个事件" % total_deleted)

# 压缩事件
func _compress_event(event: EventData):
	# 只保留核心数据
	event.timeline = []  # 删除详细时间线
	event.event_data = {}  # 删除详细数据

	# 保留摘要
	if event.observer_description.length() > 150:
		event.observer_description = event.observer_description.substr(0, 150) + "..."

# ===== 对话清理 =====

func cleanup_dialogues():
	print("[DataLifecycle] 清理对话日志...")

	# 如果有对话历史记录系统
	if DialogManager.has_method("get_dialogue_history"):
		var total_deleted = 0
		var current_time = TimeSystem.get_timestamp()

		for ai in CharacterManager.get_all_characters():
			var dialogues = DialogManager.get_dialogue_history(ai.id)

			var to_delete = []

			for dialogue in dialogues:
				var age_days = TimeSystem.calculate_duration_days(
					dialogue.timestamp,
					current_time
				)

				# 删除15天前的普通对话
				if age_days > TIER_3_RETENTION_DAYS:
					to_delete.append(dialogue.id)

			for dialogue_id in to_delete:
				DialogManager.delete_dialogue(ai.id, dialogue_id)
				total_deleted += 1

		cleanup_stats.dialogues_deleted += total_deleted
		print("[DataLifecycle] 删除了 %d 条对话记录" % total_deleted)

# ===== 天气历史清理 =====

func cleanup_weather_history():
	if WeatherSystem.has_method("cleanup_old_history"):
		WeatherSystem.cleanup_old_history(TIER_3_RETENTION_DAYS)

# ===== LLM缓存清理 =====

func cleanup_llm_cache():
	if LLMCache and LLMCache.has_method("clean_expired_cache"):
		LLMCache.clean_expired_cache()

# ===== 清理失效引用 =====

func cleanup_dead_references():
	# 清理EventBus中的失效信号连接
	# 这个需要Godot 4.x的signal.get_connections()来实现
	pass

# ===== 强制垃圾回收 =====

func force_garbage_collection():
	# Godot会自动GC,但可以提示系统
	if OS.has_feature("debug"):
		print("[DataLifecycle] 提示系统进行垃圾回收")

# ===== 手动触发清理 (开发用) =====

func manual_cleanup():
	print("[DataLifecycle] 手动触发数据清理")
	perform_full_cleanup()

# ===== 统计报告 =====

func get_cleanup_stats() -> Dictionary:
	return cleanup_stats.duplicate()

func print_stats():
	print("=== 数据清理统计 ===")
	print("总清理次数: %d" % cleanup_stats.total_cleanups)
	print("删除记忆: %d" % cleanup_stats.memories_deleted)
	print("删除事件: %d" % cleanup_stats.events_deleted)
	print("删除对话: %d" % cleanup_stats.dialogues_deleted)
```

### 记忆管理器增强

```gdscript
# 在 script/ai/memory/MemoryManager.gd 中添加

# 记忆上限
const MAX_MEMORIES_PER_AI = 100  # 每个AI最多100条记忆

# 添加记忆时自动清理
func add_memory(ai_id: String, content: String, type: MemoryType, importance: int):
	var character_data = _get_character_data(ai_id)
	var memories = character_data.get("memories", [])

	# 创建新记忆
	var memory = {
		"id": _generate_memory_id(),
		"content": content,
		"type": type,
		"importance": importance,
		"timestamp": TimeSystem.get_timestamp(),
		"created_at": Time.get_unix_time_from_system()
	}

	memories.append(memory)

	# 如果超过上限,删除最不重要的旧记忆
	if memories.size() > MAX_MEMORIES_PER_AI:
		_trim_memories(memories)

	character_data["memories"] = memories
	_save_character_data(ai_id, character_data)

# 修剪记忆
func _trim_memories(memories: Array):
	# 按重要性和时间排序
	memories.sort_custom(func(a, b):
		# 先按重要性
		if a.importance != b.importance:
			return a.importance > b.importance
		# 再按时间 (新的优先)
		return a.created_at > b.created_at
	)

	# 只保留前MAX_MEMORIES_PER_AI条
	while memories.size() > MAX_MEMORIES_PER_AI:
		var removed = memories.pop_back()
		print("[MemoryManager] 删除低重要性记忆: %s" % removed.content.substr(0, 30))

# 删除记忆 (供DataLifecycleManager调用)
func delete_memory(ai_id: String, memory_id: String):
	var character_data = _get_character_data(ai_id)
	var memories = character_data.get("memories", [])

	for i in range(memories.size()):
		if memories[i].get("id") == memory_id:
			memories.remove_at(i)
			character_data["memories"] = memories
			_save_character_data(ai_id, character_data)
			return

# 获取所有记忆 (供清理器使用)
func get_all_memories(ai_id: String) -> Array:
	var character_data = _get_character_data(ai_id)
	return character_data.get("memories", [])
```

### 对象池模式

```gdscript
# script/performance/ObjectPool.gd
class_name ObjectPool extends Node

# 对象池配置
var pools = {}

# 创建对象池
func create_pool(pool_name: String, scene_path: String, initial_size: int = 10):
	if pools.has(pool_name):
		return

	var pool = {
		"scene_path": scene_path,
		"available": [],
		"in_use": [],
		"total_created": 0
	}

	# 预创建对象
	for i in range(initial_size):
		var obj = load(scene_path).instantiate()
		obj.set_meta("pooled", true)
		pool.available.append(obj)
		pool.total_created += 1

	pools[pool_name] = pool
	print("[ObjectPool] 创建对象池: %s, 初始大小: %d" % [pool_name, initial_size])

# 获取对象
func acquire(pool_name: String) -> Node:
	if not pools.has(pool_name):
		push_error("[ObjectPool] 对象池不存在: %s" % pool_name)
		return null

	var pool = pools[pool_name]

	var obj = null

	if pool.available.size() > 0:
		# 从可用池获取
		obj = pool.available.pop_back()
	else:
		# 创建新对象
		obj = load(pool.scene_path).instantiate()
		obj.set_meta("pooled", true)
		pool.total_created += 1
		print("[ObjectPool] 池 %s 扩容,总对象: %d" % [pool_name, pool.total_created])

	pool.in_use.append(obj)
	return obj

# 归还对象
func release(pool_name: String, obj: Node):
	if not pools.has(pool_name):
		push_error("[ObjectPool] 对象池不存在: %s" % pool_name)
		return

	var pool = pools[pool_name]

	# 从使用中移除
	var idx = pool.in_use.find(obj)
	if idx != -1:
		pool.in_use.remove_at(idx)

	# 重置对象状态
	if obj.has_method("reset"):
		obj.reset()

	# 归还到可用池
	pool.available.append(obj)

# 清理对象池
func clear_pool(pool_name: String):
	if not pools.has(pool_name):
		return

	var pool = pools[pool_name]

	# 释放所有对象
	for obj in pool.available:
		obj.queue_free()
	for obj in pool.in_use:
		obj.queue_free()

	pools.erase(pool_name)
	print("[ObjectPool] 清理对象池: %s" % pool_name)

# 获取池统计
func get_pool_stats(pool_name: String) -> Dictionary:
	if not pools.has(pool_name):
		return {}

	var pool = pools[pool_name]
	return {
		"available": pool.available.size(),
		"in_use": pool.in_use.size(),
		"total_created": pool.total_created
	}
```

### 信号连接管理

```gdscript
# script/performance/SignalManager.gd
class_name SignalManager extends Node

# 追踪所有信号连接
var signal_connections = {}

# 连接信号 (带追踪)
func connect_signal(object: Object, signal_name: String, callable: Callable):
	object.connect(signal_name, callable)

	# 记录连接
	var key = "%s_%s" % [object.get_instance_id(), signal_name]
	if not signal_connections.has(key):
		signal_connections[key] = []
	signal_connections[key].append({
		"object": object,
		"signal": signal_name,
		"callable": callable
	})

# 断开所有追踪的信号
func disconnect_all_for_object(object: Object):
	var instance_id = object.get_instance_id()

	var to_remove = []

	for key in signal_connections.keys():
		if key.begins_with(str(instance_id)):
			var connections = signal_connections[key]
			for conn in connections:
				if conn.object.is_connected(conn.signal, conn.callable):
					conn.object.disconnect(conn.signal, conn.callable)
			to_remove.append(key)

	for key in to_remove:
		signal_connections.erase(key)

# 获取连接统计
func get_connection_count() -> int:
	var total = 0
	for connections in signal_connections.values():
		total += connections.size()
	return total
```

```gdscript
# script/ai/AIThrottleManager.gd
class_name AIThrottleManager extends Node

# 动态调整AI思考频率

var ai_priorities = {}  # ai_id → priority (0.0-1.0)

# 更新AI优先级
func update_ai_priority(ai_id: String):
	var ai = CharacterManager.get_character(ai_id)

	var priority = 0.5  # 默认优先级

	# 因素1: 是否被观察
	if CharacterManager.current_selected_character == ai_id:
		priority += 0.4  # 被观察的AI优先级高

	# 因素2: 是否参与重要事件
	if _is_in_important_event(ai_id):
		priority += 0.3

	# 因素3: 距离玩家相机距离
	var distance = _get_camera_distance(ai)
	if distance < 500:
		priority += 0.2

	# 限制范围 [0.1, 1.0]
	priority = clamp(priority, 0.1, 1.0)

	ai_priorities[ai_id] = priority

# 获取AI的思考间隔 (基于优先级)
func get_think_interval(ai_id: String) -> float:
	var priority = ai_priorities.get(ai_id, 0.5)

	# 高优先级: 10秒
	# 低优先级: 60秒
	var base_interval = 10.0
	var max_interval = 60.0

	return lerp(max_interval, base_interval, priority)

func _is_in_important_event(ai_id: String) -> bool:
	# 检查是否在重要事件中
	for event in EventSystem.instance.active_events.values():
		if ai_id in event.participants and event.importance >= EventData.Importance.MAJOR:
			return true
	return false

func _get_camera_distance(ai: Node) -> float:
	var camera = get_viewport().get_camera_2d()
	if camera:
		return ai.global_position.distance_to(camera.global_position)
	return 999999.0
```

### 空间分区优化

```gdscript
# script/performance/SpatialPartition.gd
class_name SpatialPartition extends Node

# 2D空间分区,用于快速查找附近AI

const CELL_SIZE = 200.0  # 每个格子200像素

var grid = {}  # {cell_key: [ai_ids]}

# 更新AI位置
func update_ai_position(ai_id: String, position: Vector2):
	# 移除旧位置
	remove_ai(ai_id)

	# 添加到新位置
	var cell_key = _get_cell_key(position)
	if not grid.has(cell_key):
		grid[cell_key] = []
	grid[cell_key].append(ai_id)

# 移除AI
func remove_ai(ai_id: String):
	for cell in grid.values():
		var idx = cell.find(ai_id)
		if idx != -1:
			cell.remove_at(idx)
			return

# 查找附近的AI
func find_nearby_ais(position: Vector2, radius: float) -> Array:
	var nearby = []

	# 计算需要检查的格子范围
	var cells_to_check = _get_cells_in_radius(position, radius)

	for cell_key in cells_to_check:
		if grid.has(cell_key):
			for ai_id in grid[cell_key]:
				var ai = CharacterManager.get_character(ai_id)
				if ai and position.distance_to(ai.global_position) <= radius:
					nearby.append(ai_id)

	return nearby

func _get_cell_key(position: Vector2) -> String:
	var x = int(position.x / CELL_SIZE)
	var y = int(position.y / CELL_SIZE)
	return "%d_%d" % [x, y]

func _get_cells_in_radius(position: Vector2, radius: float) -> Array:
	var cells = []

	var min_x = int((position.x - radius) / CELL_SIZE)
	var max_x = int((position.x + radius) / CELL_SIZE)
	var min_y = int((position.y - radius) / CELL_SIZE)
	var max_y = int((position.y + radius) / CELL_SIZE)

	for x in range(min_x, max_x + 1):
		for y in range(min_y, max_y + 1):
			cells.append("%d_%d" % [x, y])

	return cells
```

```gdscript
# script/ai/llm/LLMRequestQueue.gd
class_name LLMRequestQueue extends Node

# 并发限制
const MAX_CONCURRENT_REQUESTS = 3
const REQUEST_TIMEOUT = 10.0  # 10秒超时

var request_queue: Array = []
var active_requests: Dictionary = {}  # request_id → request_data

# 添加请求到队列
func enqueue_request(request_data: Dictionary) -> String:
	var request_id = _generate_request_id()

	request_data["id"] = request_id
	request_data["enqueued_at"] = Time.get_ticks_msec()
	request_data["priority"] = request_data.get("priority", 0.5)

	request_queue.append(request_data)

	# 按优先级排序
	request_queue.sort_custom(func(a, b):
		return a.priority > b.priority
	)

	# 尝试处理队列
	_process_queue()

	return request_id

# 处理队列
func _process_queue():
	while request_queue.size() > 0 and active_requests.size() < MAX_CONCURRENT_REQUESTS:
		var request = request_queue.pop_front()
		_execute_request(request)

# 执行请求
func _execute_request(request: Dictionary):
	active_requests[request.id] = request

	var http = await APIManager.generate_decision(request.prompt, request.character_name)

	if not http:
		_on_request_failed(request.id, "HTTP creation failed")
		return

	# 超时定时器
	var timeout_timer = Timer.new()
	timeout_timer.wait_time = REQUEST_TIMEOUT
	timeout_timer.one_shot = true
	timeout_timer.timeout.connect(func():
		_on_request_timeout(request.id)
	)
	add_child(timeout_timer)
	timeout_timer.start()

	http.request_completed.connect(func(result, response_code, headers, body):
		timeout_timer.queue_free()

		if response_code == 200:
			_on_request_completed(request.id, body.get_string_from_utf8())
		else:
			_on_request_failed(request.id, "HTTP %d" % response_code)
	)

# 请求完成
func _on_request_completed(request_id: String, response: String):
	var request = active_requests.get(request_id)
	if not request:
		return

	# 调用回调
	if request.has("callback"):
		request.callback.call(response)

	# 移除活跃请求
	active_requests.erase(request_id)

	# 继续处理队列
	_process_queue()

# 请求失败
func _on_request_failed(request_id: String, reason: String):
	var request = active_requests.get(request_id)
	if not request:
		return

	print("[LLMQueue] 请求失败: %s, 原因: %s" % [request_id, reason])

	# 重试逻辑
	request["retry_count"] = request.get("retry_count", 0) + 1

	if request.retry_count < 3:
		# 重新入队
		request_queue.insert(0, request)
	else:
		# 放弃,调用失败回调
		if request.has("error_callback"):
			request.error_callback.call(reason)

	active_requests.erase(request_id)
	_process_queue()

# 请求超时
func _on_request_timeout(request_id: String):
	_on_request_failed(request_id, "Timeout")

# 获取队列统计
func get_queue_stats() -> Dictionary:
	return {
		"queue_length": request_queue.size(),
		"active_requests": active_requests.size(),
		"max_concurrent": MAX_CONCURRENT_REQUESTS
	}

func _generate_request_id() -> String:
	return "req_%d" % Time.get_ticks_msec()
```

```gdscript
# script/save/SaveCompressor.gd
class_name SaveCompressor extends Node

# 压缩存档数据
func compress_save_data(data: Dictionary) -> PackedByteArray:
	# 1. 转为JSON字符串
	var json_string = JSON.stringify(data)

	# 2. 转为UTF-8字节
	var json_bytes = json_string.to_utf8_buffer()

	# 3. GZIP压缩
	var compressed = json_bytes.compress(FileAccess.COMPRESSION_GZIP)

	var original_size = json_bytes.size()
	var compressed_size = compressed.size()
	var ratio = float(compressed_size) / float(original_size) * 100.0

	print("[SaveCompressor] 压缩: %d → %d bytes (%.1f%%)" % [
		original_size,
		compressed_size,
		ratio
	])

	return compressed

# 解压存档数据
func decompress_save_data(compressed: PackedByteArray) -> Dictionary:
	# 1. GZIP解压
	var decompressed = compressed.decompress_dynamic(-1, FileAccess.COMPRESSION_GZIP)

	# 2. 转为字符串
	var json_string = decompressed.get_string_from_utf8()

	# 3. 解析JSON
	var json = JSON.new()
	var parse_result = json.parse(json_string)

	if parse_result != OK:
		push_error("[SaveCompressor] 解压失败")
		return {}

	return json.get_data()
```

### 增量存档

```gdscript
# script/save/IncrementalSaveManager.gd
class_name IncrementalSaveManager extends Node

# 上次完整存档
var last_full_save: Dictionary = {}
var last_full_save_time: float = 0.0

const FULL_SAVE_INTERVAL = 3600.0  # 每小时完整存档一次

# 保存 (智能选择增量或完整)
func save_game(save_name: String):
	var current_time = Time.get_unix_time_from_system()

	# 判断是否需要完整存档
	if current_time - last_full_save_time >= FULL_SAVE_INTERVAL:
		_save_full(save_name)
	else:
		_save_incremental(save_name)

# 完整存档
func _save_full(save_name: String):
	print("[IncrementalSave] 执行完整存档...")

	var save_data = GameSaveManager.instance._collect_all_game_data(save_name, "manual")

	# 压缩并保存
	var compressed = SaveCompressor.compress_save_data(save_data)

	var file = FileAccess.open("user://saves/%s.sav" % save_name, FileAccess.WRITE)
	if file:
		file.store_buffer(compressed)
		file.close()

	# 更新基准
	last_full_save = save_data.duplicate(true)
	last_full_save_time = Time.get_unix_time_from_system()

	print("[IncrementalSave] 完整存档完成")

# 增量存档 (只保存变化)
func _save_incremental(save_name: String):
	print("[IncrementalSave] 执行增量存档...")

	var current_data = GameSaveManager.instance._collect_all_game_data(save_name, "auto")

	# 计算差异
	var delta = _calculate_delta(last_full_save, current_data)

	# 保存差异
	var file = FileAccess.open("user://saves/%s.inc" % save_name, FileAccess.WRITE)
	if file:
		var compressed = SaveCompressor.compress_save_data(delta)
		file.store_buffer(compressed)
		file.close()

	print("[IncrementalSave] 增量存档完成,变化量: %d bytes" % JSON.stringify(delta).length())

# 计算数据差异
func _calculate_delta(old_data: Dictionary, new_data: Dictionary) -> Dictionary:
	var delta = {}

	for key in new_data.keys():
		if not old_data.has(key):
			# 新增数据
			delta[key] = new_data[key]
		elif typeof(new_data[key]) == TYPE_DICTIONARY:
			# 递归比较字典
			var sub_delta = _calculate_delta(old_data[key], new_data[key])
			if sub_delta.size() > 0:
				delta[key] = sub_delta
		elif old_data[key] != new_data[key]:
			# 值变化
			delta[key] = new_data[key]

	return delta
```

```gdscript
extends Node

var metrics = {
```
