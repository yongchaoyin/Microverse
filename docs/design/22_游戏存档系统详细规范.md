# 22. 游戏存档系统详细规范 (Game Save System Specification)

## 📋 系统概述

游戏存档系统是Microverse的**数据持久化引擎**,负责将游戏世界的完整状态序列化保存,并能在任意时刻完整恢复。由于Microverse有11个复杂的子系统(时间、经济、性格、关系、记忆、日程、任务、天气、导航、冲突、LLM),存档系统需要精心设计以确保数据完整性和版本兼容性。

**设计哲学:**
- 存档要**完整无损** - 加载后游戏状态100%恢复
- 存档要**向前兼容** - 旧版本存档能在新版本加载
- 存档要**可读可调试** - 使用JSON格式,便于查看和修改
- 存档要**安全可靠** - 防止数据损坏,支持备份和恢复

**核心功能:**
- 自动存档 (每游戏日自动保存)
- 手动存档 (玩家随时保存)
- 快速存档/快速读档 (F5/F9)
- 多存档槽位 (支持10个独立存档)
- 存档元数据 (时间、游戏进度、截图)
- 版本迁移系统
- 云存档支持 (可选)

## 二、存档系统架构

### 2.1 SaveManager核心类

```gdscript
# script/save/SaveManager.gd
extends Node

class_name SaveManager

# ========================================
# 常量
# ========================================

const SAVE_VERSION = "1.2.3"
const SAVE_DIR = "user://saves/"
const SCREENSHOT_DIR = "user://saves/screenshots/"
const BACKUP_DIR = "user://saves/backups/"
const AUTOSAVE_DIR = "user://saves/autosave/"

const MAX_SAVE_SLOTS = 10
const MAX_AUTOSAVE_COUNT = 5
const MAX_BACKUP_COUNT = 3

# ========================================
# 信号
# ========================================

signal save_started()
signal save_completed(success: bool, save_name: String)
signal save_failed(error_message: String)

signal load_started()
signal load_completed(success: bool, save_name: String)
signal load_failed(error_message: String)

signal save_list_updated(save_files: Array)

# ========================================
# 数据
# ========================================

var current_save_name: String = ""
var is_saving: bool = false
var is_loading: bool = false

# ========================================
# 初始化
# ========================================

func _ready():
    print("[SaveManager] 存档系统初始化")
    _ensure_directories_exist()

    # 连接时间系统 (自动存档)
    if TimeSystem:
        TimeSystem.day_changed.connect(_on_day_changed)

func _ensure_directories_exist():
    """确保所有存档目录存在"""
    var directories = [SAVE_DIR, SCREENSHOT_DIR, BACKUP_DIR, AUTOSAVE_DIR]

    for dir_path in directories:
        var dir = DirAccess.open("user://")
        if not dir.dir_exists(dir_path):
            var result = dir.make_dir_recursive(dir_path)
            if result == OK:
                print("  - 创建目录: %s" % dir_path)
            else:
                push_error("  - 无法创建目录: %s" % dir_path)

# ========================================
# 保存游戏
# ========================================

func save_game(save_name: String = "", save_type: String = "manual", create_screenshot: bool = true) -> bool:
    """保存游戏"""

    if is_saving:
        push_warning("[SaveManager] 正在保存中,请稍候")
        return false

    is_saving = true
    save_started.emit()

    # 生成存档名称
    if save_name.is_empty():
        save_name = _generate_save_name(save_type)

    print("[SaveManager] 开始保存游戏: %s" % save_name)

    # 收集游戏数据
    var save_data = await _collect_all_game_data(save_name, save_type)

    # 创建截图
    if create_screenshot:
        var screenshot_path = await _capture_screenshot(save_name)
        save_data["save_meta"]["screenshot_path"] = screenshot_path

    # 写入存档文件
    var file_path = _get_save_path(save_name, save_type)
    var success = _write_save_file(file_path, save_data)

    if success:
        current_save_name = save_name
        print("[SaveManager] 游戏保存成功: %s" % file_path)
        save_completed.emit(true, save_name)

        # 创建备份
        _create_backup(save_name, save_type)
    else:
        push_error("[SaveManager] 游戏保存失败: %s" % file_path)
        save_failed.emit("保存文件写入失败")

    is_saving = false
    return success

func _collect_all_game_data(save_name: String, save_type: String) -> Dictionary:
    """收集所有游戏数据"""

    print("  [1/12] 收集元数据...")
    var save_data = {
        "save_meta": _collect_save_meta(save_name, save_type),
    }

    print("  [2/12] 收集时间系统数据...")
    save_data["time_system"] = _collect_time_system_data()

    print("  [3/12] 收集天气系统数据...")
    save_data["weather_system"] = _collect_weather_system_data()

    print("  [4/12] 收集角色数据...")
    save_data["characters"] = await _collect_all_characters_data()

    print("  [5/12] 收集地点数据...")
    save_data["locations"] = _collect_locations_data()

    print("  [6/12] 收集全局事件数据...")
    save_data["global_events"] = _collect_global_events_data()

    print("  [7/12] 收集冲突历史数据...")
    save_data["conflict_history"] = _collect_conflict_history_data()

    print("  [8/12] 收集关系网络数据...")
    save_data["relationship_network"] = _collect_relationship_network_data()

    print("  [9/12] 收集任务系统数据...")

func load_game(save_name: String, save_type: String = "manual") -> bool:
    """加载游戏"""

    if is_loading:
        push_warning("[SaveManager] 正在加载中,请稍候")
        return false

    is_loading = true
    load_started.emit()

    var file_path = _get_save_path(save_name, save_type)

    if not FileAccess.file_exists(file_path):
        push_error("[SaveManager] 存档文件不存在: %s" % file_path)
        load_failed.emit("存档文件不存在")
        is_loading = false
        return false

    print("[SaveManager] 开始加载游戏: %s" % save_name)

    # 读取存档文件
    var save_data = _read_save_file(file_path)
    if not save_data:
        push_error("[SaveManager] 存档文件读取失败")
        load_failed.emit("存档文件损坏")
        is_loading = false
        return false

    # 验证版本兼容性
    if not _validate_save_version(save_data):
        push_warning("[SaveManager] 存档版本不兼容,尝试迁移...")
        save_data = _migrate_save_data(save_data)

    # 应用游戏数据
    var success = await _apply_all_game_data(save_data)

    if success:
        current_save_name = save_name
        print("[SaveManager] 游戏加载成功")
        load_completed.emit(true, save_name)
    else:
        push_error("[SaveManager] 游戏加载失败")
        load_failed.emit("数据应用失败")

    is_loading = false
    return success

func _read_save_file(file_path: String) -> Dictionary:
    """读取存档文件"""
    var file = FileAccess.open(file_path, FileAccess.READ)
    if not file:
        return {}

    var content = file.get_as_text()
    file.close()

    if content.strip_edges().is_empty():
        return {}

    var data = JSON.parse_string(content)
    if not data:
        return {}

    return data

        save_data = _migrate_to_v1_1_0(save_data)

    # v1.1.0 → v1.2.0
    if _version_less_than(old_version, "1.2.0"):
        print("    - 应用 v1.2.0 迁移...")
        save_data = _migrate_to_v1_2_0(save_data)

    save_data["save_meta"]["version"] = current_version
    save_data["save_meta"]["migrated_from"] = old_version
    save_data["save_meta"]["migration_timestamp"] = Time.get_unix_time_from_system()

    return save_data

func _migrate_to_v1_1_0(data: Dictionary) -> Dictionary:
    """迁移到v1.1.0"""
    # 添加天气系统数据
    if not data.has("weather_system"):
        data["weather_system"] = {
            "current_weather": "sunny",
            "weather_intensity": 1.0,
            "weather_duration_remaining": 6.0,
            "last_weather_check_hour": 0.0
        }

    return data

func _migrate_to_v1_2_0(data: Dictionary) -> Dictionary:
    """迁移到v1.2.0"""
    # 添加冲突历史
    if not data.has("conflict_history"):
        data["conflict_history"] = []

    # 添加LLM缓存
    if not data.has("llm_cache"):
        data["llm_cache"] = {
            "prompt_cache": {},
            "conversation_summaries": {},
            "llm_usage_stats": {
                "total_calls": 0,
                "total_tokens_used": 0,
                "total_cost_usd": 0.0
            }
        }

    return data

func _version_less_than(version_a: String, version_b: String) -> bool:
    """比较版本号"""
    var parts_a = version_a.split(".")
    var parts_b = version_b.split(".")

    for i in range(3):
        var a = int(parts_a[i]) if i < parts_a.size() else 0
        var b = int(parts_b[i]) if i < parts_b.size() else 0

        if a < b:
            return true
        elif a > b:
            return false

    return false
```
