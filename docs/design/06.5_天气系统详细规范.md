# 06.5 天气系统详细规范 (Weather System Specification)

## 📋 系统概述

天气系统为Microverse增添**真实的环境氛围感**,天气不仅是视觉效果,更是影响AI行为、情绪、活动选择的重要环境因素。

**设计哲学:**
- 天气是**动态而非静态** - 一天内天气可以变化
- 天气影响**AI决策** - 下雨天更可能待在室内,晴天更愿意外出
- 天气创造**氛围感** - 配合音效、粒子效果、光照变化
- 天气要**符合季节** - 春多雨,夏晴朗,秋凉爽,冬多雪

**核心功能:**
- 7种天气类型 (晴/阴/雨/雷暴/雪/雾/大风)
- 动态天气转换系统
- 季节-天气概率关联
- 天气对AI行为的多维度影响
- 视觉/音效/光照效果

## 🔄 动态天气转换系统

### 天气转换规则

```yaml
天气转换模型:
  每6小时游戏时间 (36分钟现实) 重新计算天气:
    - 检查当前季节的天气概率分布
    - 考虑当前天气的转换倾向
    - 生成新天气

  天气持续时间:
    - 最短: 3小时 (18分钟现实)
    - 最长: 12小时 (72分钟现实)
    - 平均: 6小时

  天气转换概率矩阵:
    晴朗 →:
      晴朗: 60%
      多云: 30%
      小雨: 8%
      大风: 2%

    多云 →:
      多云: 40%
      晴朗: 30%
      小雨: 20%
      大雨: 5%
      雷暴: 3%
      大雾: 2%

    小雨 →:
      小雨: 40%
      多云: 30%
      大雨: 20%
      雷暴: 5%
      晴朗: 5%

    大雨 →:
      大雨: 50%
      雷暴: 20%
      小雨: 20%
      多云: 10%

    雷暴 →:
      雷暴: 30%
      大雨: 50%
      小雨: 15%
      多云: 5%

    下雪 → (仅冬季):
      下雪: 60%
      多云: 30%
      晴朗: 10%

    大雾 →:
      大雾: 40%
      多云: 40%
      晴朗: 20%

    大风 →:
      大风: 30%
      晴朗: 40%
      多云: 30%

  特殊规则:
    - 春季: 晴→雨转换概率×1.5
    - 夏季: 保持晴天概率×1.3
    - 秋季: 晴→多云概率×1.2
    - 冬季: 只有晴/多云/雪/雾,无雨雷暴
```

func _on_hour_changed(new_hour: int):
    """每小时检查天气"""
    var current_hour = TimeSystem.get_current_hour() if TimeSystem else 0.0

    # 减少持续时间
    weather_duration_remaining -= 1.0

    # 检查是否需要更新天气
    if weather_duration_remaining <= 0 or (current_hour - last_weather_check_hour) >= WEATHER_UPDATE_INTERVAL:
        _update_weather()
        last_weather_check_hour = current_hour

func _on_season_changed(new_season: String):
    """季节变化时重新评估天气"""
    print("[WeatherSystem] 季节变化到 %s, 重新生成天气" % new_season)
    _update_weather()

func _update_weather():
    """更新天气"""
    var old_weather = current_weather
    var season = TimeSystem.current_season if TimeSystem else "Spring"

    # 根据当前天气和季节生成新天气
    var new_weather = _generate_next_weather(old_weather, season)

    if new_weather != old_weather:
        current_weather = new_weather
        weather_duration_remaining = randf_range(3.0, 12.0)
        weather_changed.emit(new_weather, old_weather)

        print("[WeatherSystem] 天气变化: %s → %s (持续约%.1f小时)" % [
            WEATHER_DATA[old_weather].name_cn,
            WEATHER_DATA[new_weather].name_cn,
            weather_duration_remaining
        ])

        # 应用天气效果
        _apply_weather_effects()

    }

    var transitions = transition_matrix.get(current, {"sunny": 1.0})

    # 应用季节修正
    transitions = _apply_season_modifiers(transitions, season)

    # 加权随机选择
    return _weighted_random_choice(transitions)

func _pick_weather_by_season(season: String) -> String:
    """根据季节概率选择天气"""
    var season_probabilities = {
        "Spring": {"sunny": 0.6, "cloudy": 0.1, "light_rain": 0.25, "foggy": 0.05},
        "Summer": {"sunny": 0.8, "cloudy": 0.1, "light_rain": 0.05, "windy": 0.05},
        "Autumn": {"sunny": 0.7, "cloudy": 0.2, "light_rain": 0.05, "foggy": 0.05},
        "Winter": {"sunny": 0.3, "cloudy": 0.2, "snowy": 0.4, "foggy": 0.1}
    }

    var probabilities = season_probabilities.get(season, {"sunny": 1.0})
    return _weighted_random_choice(probabilities)

func _pick_winter_weather() -> String:
    """冬季天气"""
    var options = {"snowy": 0.5, "cloudy": 0.3, "sunny": 0.15, "foggy": 0.05}
    return _weighted_random_choice(options)

func _apply_season_modifiers(transitions: Dictionary, season: String) -> Dictionary:
    """应用季节修正"""
    var modified = transitions.duplicate()

    match season:
        "Spring":
            # 春季雨水增多
            if modified.has("light_rain"):
                modified["light_rain"] *= 1.5
            if modified.has("heavy_rain"):
                modified["heavy_rain"] *= 1.3
        "Summer":
            # 夏季保持晴天
            if modified.has("sunny"):
                modified["sunny"] *= 1.3
            if modified.has("light_rain"):
                modified["light_rain"] *= 0.7
        "Autumn":
            # 秋季多云
            if modified.has("cloudy"):
                modified["cloudy"] *= 1.2
        "Winter":
            # 冬季移除雨雷暴
            modified.erase("light_rain")
            modified.erase("heavy_rain")
            modified.erase("thunderstorm")

    # 重新归一化概率
    var total = 0.0
    for prob in modified.values():
        total += prob

    for key in modified.keys():
        modified[key] /= total

    return modified

func _weighted_random_choice(options: Dictionary) -> String:
    """加权随机选择"""
    var rand_val = randf()
    var cumulative = 0.0

    for weather_type in options.keys():
        cumulative += options[weather_type]
        if rand_val <= cumulative:
            return weather_type

    return options.keys()[0]  # 默认返回第一个
