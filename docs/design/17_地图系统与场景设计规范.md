## 1. 设计哲学与视觉风格

### 1.1 核心设计原则

```
Microverse地图设计五大原则:
1. 现代西方小镇氛围 - 温馨宜居的现代小镇,生活气息浓厚
2. Stardew Valley美学 - 像素艺术+俯视2.5D视角+温暖色调
3. 观察者友好 - 清晰的视觉层次,便于观察AI活动
4. 生活化场景 - 8个AI的日常生活空间(家、办公室、咖啡馆等)
5. 有机扩展 - 支持未来添加新场景和功能区域
```

### 1.2 视觉风格规范

```
【像素艺术风格】
- Tile尺寸: 16×16像素(基础单位)
- 角色尺寸: 16×32像素(1×2 tiles)
- 建筑尺寸: 多tile组合(如办公室 40×30 tiles)
- 手绘感 + 像素化,避免过度抗锯齿

【俯视视角】
- 2.5D俯视角度(类似Stardew Valley)
- 轻微透视感,垂直物体有深度
- 墙壁使用等距投影表现立体感

【色彩风格】
- 主色调: 温暖的棕色、米色、淡绿(现代小镇温馨感)
- 辅助色: 蓝色(天空/水)、绿色(植物)、红色(砖墙)
- 饱和度: 适中偏明亮,星露谷式的温馨色彩
- 昼夜变化: 使用CanvasModulate调节色温

【动画规范】
- 水面波光: 4帧循环,0.3秒/帧
- 树叶摇摆: 3帧循环,0.5秒/帧
- 角色走路: 4帧循环,0.15秒/帧
- 其他装饰: 2-4帧,保持自然节奏
```

```gdscript

草地类 (Grass):

- 磨损效果: 随机裂缝

水泥地 (Concrete):
Tile ID: 48-63
- 干净水泥: 4个变体
- 旧水泥: 4个变体(裂纹)
- 水泥+草边缘: 8个过渡tile
颜色: #BDBDBD (灰色)

泥土路 (Dirt):
Tile ID: 64-79
- 干泥土: 4个变体
- 湿泥土: 4个变体
- 泥土+石头: 4个变体
颜色: #8B6F47 (棕色)

木质地板 (Wood Floor - 室内):
Tile ID: 80-95
- 木地板: 纵向纹理
- 地毯: 不同颜色
颜色: #D4A373 (温暖木色)
```

#### 3.1.2 Auto-tiling配置

```gdscript
# ground_tileset.tres (Godot TileSet资源)

# 草地Auto-tile配置
[sub_resource type="TileSet" id=1]

# Tile 0: 草地(使用Blob 47-tile规则)
0/name = "grass"
0/texture = ExtResource("ground.png")
0/tex_offset = Vector2(0, 0)
0/modulate = Color(1, 1, 1, 1)
0/region = Rect2(0, 0, 16, 16)
0/tile_mode = 1 # Auto-tile模式
0/autotile/bitmask_mode = 1 # 2×2模式
0/autotile/tile_size = Vector2(16, 16)
0/autotile/spacing = 0

# Bitmask配置(47个tile位置)
0/autotile/bitmask_flags = [
    Vector2(0, 0), 0,    # 完全被包围
    Vector2(1, 0), 1,    # 右边缘
    Vector2(2, 0), 2,    # 左边缘
    Vector2(0, 1), 4,    # 下边缘
    Vector2(0, 2), 8,    # 上边缘
    Vector2(1, 1), 5,    # 右下角
    Vector2(2, 1), 6,    # 左下角
    Vector2(1, 2), 9,    # 右上角
    Vector2(2, 2), 10,   # 左上角
    # ... (共47个配置)
]

# 碰撞形状(草地可通行)
0/shapes = []

# Tile 1: 道路(Auto-tile)
1/name = "road"
1/texture = ExtResource("ground.png")
1/region = Rect2(0, 16, 16, 16)
1/tile_mode = 1
1/autotile/bitmask_mode = 1
1/autotile/tile_size = Vector2(16, 16)
# ... (类似草地配置)

# 碰撞形状(道路可通行但影响移动速度)
1/shapes = []
```

#### 3.1.3 水面动画Tileset

```gdscript
【Water Tileset】 - res://assets/tilesets/water.png (256×256)

湖水/河流 (4帧动画):
Frame 1: 平静水面 + 波纹1
Frame 2: 平静水面 + 波纹2
Frame 3: 平静水面 + 波纹3
Frame 4: 平静水面 + 波纹4

颜色:
- 浅水: #6DAEDB (浅蓝)
- 深水: #4287C8 (深蓝)
- 波光: 白色高光 #FFFFFF (20%透明度)

# water_tileset.tres
[sub_resource type="TileSet" id=2]

2/name = "water_shallow"
2/texture = ExtResource("water.png")
2/region = Rect2(0, 0, 64, 16) # 4帧水平排列
2/tile_mode = 2 # 动画模式
2/autotile/tile_size = Vector2(16, 16)

# 动画配置
2/animation/frames = 4
2/animation/frame_duration = 0.3 # 0.3秒/帧
2/animation/frame_0 = Rect2(0, 0, 16, 16)
2/animation/frame_1 = Rect2(16, 0, 16, 16)
2/animation/frame_2 = Rect2(32, 0, 16, 16)
2/animation/frame_3 = Rect2(48, 0, 16, 16)

# 碰撞形状(水不可通行)
2/shapes = [
    {
        "shape": SubResource("ConvexPolygonShape2D", 1),
        "one_way": false
    }
]
```

### 4.3 角色动画实现

```gdscript
# res://scene/character/AICharacter.gd (动画控制部分)

extends CharacterBody2D
class_name AICharacter

@onready var animated_sprite = $AnimatedSprite2D
@onready var shadow = $Shadow  # 脚下阴影

var current_direction = "down"
var is_moving = false

func _ready():
	# 加载Sprite Sheet
	animated_sprite.sprite_frames = load("res://assets/characters/%s_frames.tres" % character_data.id)

	# 设置默认动画
	animated_sprite.play("idle_down")

	# 阴影设置
	shadow.texture = preload("res://assets/effects/shadow.png")
	shadow.modulate = Color(0, 0, 0, 0.3)

func _physics_process(delta):
	# 从AI系统获取移动方向
	var direction = Vector2.ZERO
	if character_data.current_data.has("movement_target"):
		direction = (character_data.current_data.movement_target - global_position).normalized()

	# 移动
	if direction.length() > 0:
		velocity = direction * character_data.stats.get("movement_speed", 100)
		is_moving = true
		update_direction(direction)
	else:
		velocity = Vector2.ZERO
		is_moving = false

	move_and_slide()

	# 更新动画
	update_animation()

func update_direction(dir: Vector2):
	"""根据移动方向更新朝向"""
	if abs(dir.x) > abs(dir.y):
		current_direction = "right" if dir.x > 0 else "left"
	else:
		current_direction = "down" if dir.y > 0 else "up"

func update_animation():
	"""更新动画状态"""
	var anim_name = ""

	if is_moving:
		anim_name = "walk_" + current_direction
	else:
		anim_name = "idle_" + current_direction

	if animated_sprite.animation != anim_name:
		animated_sprite.play(anim_name)

# 特殊动作动画
func play_action(action_name: String):
	"""播放特殊动作(坐下、工作等)"""
	animated_sprite.play(action_name)
	await animated_sprite.animation_finished
	# 恢复默认动画
	update_animation()
```

```gdscript
extends Camera2D

# 拖拽控制

var drag_start_pos = Vector2.ZERO

# 地图边界
var map_bounds = Rect2(0, 0, 2400, 1920)  # 150×120 tiles

func _ready():
	# 设置初始缩放
	zoom = Vector2(1.0, 1.0)

	# 设置相机限制
	limit_left = int(map_bounds.position.x)
	limit_top = int(map_bounds.position.y)
	limit_right = int(map_bounds.position.x + map_bounds.size.x)
	limit_bottom = int(map_bounds.position.y + map_bounds.size.y)

	position_smoothing_speed = smooth_speed

	if follow_mode and target_character:
		# 平滑跟随角色
		var target_pos = target_character.global_position
		global_position = global_position.lerp(target_pos, 0.1)

	# 拖拽移动
	if is_dragging:
		follow_mode = false  # 拖拽时取消跟随

func _input(event):
	# 鼠标滚轮缩放
	if event is InputEventMouseButton:
		if event.button_index == MOUSE_BUTTON_WHEEL_UP and event.pressed:
			zoom_in()
		elif event.button_index == MOUSE_BUTTON_WHEEL_DOWN and event.pressed:
			zoom_out()

		# 中键拖拽
		elif event.button_index == MOUSE_BUTTON_MIDDLE:
			if event.pressed:
				is_dragging = true
				drag_start_pos = get_global_mouse_position()
			else:
				is_dragging = false

	# 鼠标移动(拖拽)
	if event is InputEventMouseMotion and is_dragging:
		var mouse_pos = get_global_mouse_position()
		var drag_offset = drag_start_pos - mouse_pos
		global_position += drag_offset

	# 键盘快捷键
	if event.is_action_pressed("camera_reset"):  # R键
		reset_camera()
	elif event.is_action_pressed("camera_toggle_follow"):  # F键
		toggle_follow_mode()

func zoom_in():
	"""放大"""
	if current_zoom_index < zoom_levels.size() - 1:
		current_zoom_index += 1
		var tween = create_tween()
		tween.tween_property(self, "zoom", Vector2(zoom_levels[current_zoom_index], zoom_levels[current_zoom_index]), 0.2)

func zoom_out():
	"""缩小"""
	if current_zoom_index > 0:
		current_zoom_index -= 1
		var tween = create_tween()
		tween.tween_property(self, "zoom", Vector2(zoom_levels[current_zoom_index], zoom_levels[current_zoom_index]), 0.2)

func reset_camera():
	"""重置相机"""
	current_zoom_index = 2
	zoom = Vector2(1.0, 1.0)
	if target_character:
		global_position = target_character.global_position
	follow_mode = true

func toggle_follow_mode():
	"""切换跟随模式"""
	follow_mode = !follow_mode

func follow_character(character: AICharacter):
	"""跟随指定角色"""
	target_character = character
	follow_mode = true

func _on_focus_changed(ai_id: String):
	"""UI焦点变化时自动跟随"""
	var character = CharacterManager.get_character_by_id(ai_id)
	if character:
		follow_character(character)
```

```gdscript
# res://script/world/InteractionZone.gd

extends Area2D
class_name InteractionZone

@export var zone_name: String = "UnnamedZone"

	# 连接信号
	body_entered.connect(_on_body_entered)
	body_exited.connect(_on_body_exited)
	mouse_entered.connect(_on_mouse_entered)
	mouse_exited.connect(_on_mouse_exited)

	"""角色进入区域"""
	if body is AICharacter:
		characters_in_zone.append(body)

		# 自动触发交互(根据AI决策)
		if enabled:
			_trigger_interaction(body)

	"""鼠标悬停(观察者视角)"""

		highlight_sprite.visible = true

	"""鼠标离开"""
	if highlight_sprite:
		highlight_sprite.visible = false

	"""触发交互"""
	match interaction_type:
		"enter":
			# 进入建筑
			_enter_building(character)
		"work":
			# 开始工作
			_start_working(character)
		"sit":
			# 坐下
			_sit_down(character)
		"talk":
			# 对话区域
			_talk_zone(character)

func _enter_building(character: AICharacter):
	"""进入建筑(切换场景)"""
	print("[InteractionZone] %s 进入 %s" % [character.character_data.character_name, zone_name])

	# 发送事件
	interaction_triggered.emit(zone_name, character)

	# 切换到室内场景(未来实现)
	# SceneManager.load_interior(zone_name, character)

func _start_working(character: AICharacter):
	"""开始工作"""
	print("[InteractionZone] %s 在 %s 工作" % [character.character_data.character_name, zone_name])

	# 播放工作动画
	character.play_action("work")

	# 通知AI系统
	EventBus.emit_signal("work_started", character.character_data.id)

func _sit_down(character: AICharacter):
	"""坐下"""
	print("[InteractionZone] %s 坐下休息" % character.character_data.character_name)

	# 播放坐下动画
	character.play_action("sit")

func _talk_zone(character: AICharacter):
	"""对话区域"""
	# 检查是否有其他AI在附近
	for other in characters_in_zone:
		if other != character and other is AICharacter:
			# 触发对话
			_start_dialogue(character, other)
			break
```

```gdscript
		entrance_zone.interaction_triggered.connect(_on_entrance_triggered)

	"""有角色进入建筑"""
	print("[Building] %s 进入 %s" % [character.character_data.character_name, building_name])

		"house":
			_enter_house(character)
		"office":
			_enter_office(character)
		"cafe":

		"bar":

	"""进入住宅"""
	# 检查是否是自己的家
	if character.character_data.id == owner_id:
		print("[Building] %s 回到自己的家" % character.character_data.character_name)
		# 切换到室内场景
		load_interior_scene("house_interior", character)
	else:
		print("[Building] %s 拜访 %s 的家" % [character.character_data.character_name, owner_id])
		# 访客进入(检查关系、时间等)
		if _can_visit(character):
			load_interior_scene("house_interior", character)
		else:
			print("[Building] 现在不方便访问")

func _can_visit(character: AICharacter) -> bool:
	"""检查是否可以访问"""
	# 检查时间(深夜不能访问)
	if TimeSystem.current_hour < 8 or TimeSystem.current_hour > 22:
		return false

	# 检查关系(好感度是否足够)
	var relationship = RelationshipManager.get_relationship(character.character_data.id, owner_id)
	if relationship.relationship_level in ["friend", "close_friend", "best_friend", "romantic"]:
		return true

	return false
```

## 10. 性能优化

### 10.1 视野裁剪

```gdscript
# res://script/world/ViewportCulling.gd

extends Node

# 相机可见范围(加上缓冲区)
var visible_rect = Rect2()
var buffer_tiles = 5  # 缓冲区(避免突然出现)

func _process(delta):
	# 获取相机视野
	var camera = get_viewport().get_camera_2d()
	if camera:
		var viewport_size = get_viewport().get_visible_rect().size
		var camera_pos = camera.global_position
		var zoom = camera.zoom.x

		# 计算可见矩形
		var half_size = viewport_size / (2 * zoom)
		visible_rect = Rect2(
			camera_pos - half_size - Vector2(buffer_tiles * 16, buffer_tiles * 16),
			viewport_size / zoom + Vector2(buffer_tiles * 32, buffer_tiles * 32)
		)

		# 裁剪场景对象
		cull_objects()

func cull_objects():
	"""裁剪不可见对象"""
	# 裁剪树木
	for tree in get_tree().get_nodes_in_group("trees"):
		if visible_rect.has_point(tree.global_position):
			tree.visible = true
			tree.process_mode = Node.PROCESS_MODE_INHERIT
		else:
			tree.visible = false
			tree.process_mode = Node.PROCESS_MODE_DISABLED

	# 裁剪装饰物
	for decor in get_tree().get_nodes_in_group("decorations"):
		if visible_rect.has_point(decor.global_position):
			decor.visible = true
		else:
			decor.visible = false
```
