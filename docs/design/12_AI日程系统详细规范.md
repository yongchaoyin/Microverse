# AI日程系统详细规范 (Schedule System)

## 📋 系统概述

AI日程系统是Microverse的**行为调度引擎**。每个AI拥有动态生成的日程表，日程由**职业模板**、**性格特征**、**关系网络**、**记忆经历**共同决定，并通过**LLM实时调整**以应对突发事件。

**设计原则**:
- 日程是**多层次的**（固定模板 + 动态生成 + 实时调整）
- 日程是**个性化的**（不同性格AI有不同的日程偏好）
- 日程是**社交驱动的**（朋友邀约、恋人约会会改变计划）
- 日程是**涌现的**（AI之间的日程冲突和巧合创造故事）

## 🏗️ 日程系统架构

### 三层日程结构

```yaml
日程系统层次:
  第一层 - 固定日程模板 (Fixed Schedule Template):
    来源: 职业系统定义
    优先级: P0 (最高)
    特点: 工作日固定时间、固定地点
    示例: 前端工程师周一至五 9:00-18:00 在办公室

  第二层 - 周期性日程 (Recurring Schedule):
    来源: AI习惯、社交关系
    优先级: P1
    特点: 每周固定的社交活动
    示例:
      - 每周三下午茶和Alice见面
      - 每周六早上去咖啡馆
      - 每月第一个周五发工资后去购物

  第三层 - 动态日程 (Dynamic Schedule):
    来源: LLM决策、社交邀约、随机事件
    优先级: P2
    特点: 实时生成、可被取消
    示例:
      - Chloe邀请我今晚去酒吧
      - 今天心情不好，想独处
      - 生病了，取消所有社交活动
```

## 🧠 LLM驱动的日程生成

### 动态日程生成流程

```gdscript
func generate_daily_schedule(ai: Node, day_type: String) -> Array:
    """
    为AI生成当天的完整日程

    参数:
      - day_type: "weekday" | "weekend"

    流程:
      1. 加载固定日程模板（工作、睡眠、饮食）
      2. 加载周期性日程（固定社交）
      3. 生成动态日程（基于性格、关系、记忆）
      4. 合并并解决冲突
      5. 返回最终日程
    """

    var schedule = []

    # 第一步: 固定模板
    if day_type == "weekday":
        schedule.append_array(_load_work_schedule(ai))
    schedule.append_array(_load_basic_life_schedule(ai))

    # 第二步: 周期性日程
    schedule.append_array(_load_recurring_schedule(ai))

    # 第三步: 动态生成
    var free_slots = _calculate_free_time_slots(schedule)
    for slot in free_slots:
        var activity = _generate_dynamic_activity(ai, slot, day_type)
        if activity:
            schedule.append(activity)

    # 第四步: 冲突解决
    schedule = _resolve_schedule_conflicts(schedule)

    # 第五步: 排序
    schedule.sort_custom(func(a, b): return a.start_hour < b.start_hour)

    return schedule
```

```gdscript
func _generate_dynamic_activity(ai: Node, time_slot: Dictionary, day_type: String) -> ScheduleEntry:
    """
    使用LLM生成动态活动

    time_slot = {start_hour: 19.0, end_hour: 22.0, duration: 3.0}
    """

    # 构建LLM提示
    var personality_desc = _format_personality(ai.personality)
    var mood = ai.current_mood
    var recent_memories = MemorySystem.build_llm_context(ai, "daily_summary", 500)
    var friends = RelationshipSystem.get_all_friends(ai.id, 40)
    var friends_str = ", ".join(friends) if friends.size() > 0 else "暂无亲密朋友"

    var prompt = """
你是%s，今天是%s。

【性格】
%s

【当前心情】%d/100

【近期记忆】
%s

【朋友列表】%s

【空闲时间】%d:00 - %d:00 (共%.1f小时)

请决定你会在这段时间做什么。选项包括:
1. 独处放松（在家、图书馆、公园）
2. 主动约朋友见面（如果有朋友且关系好）
3. 去咖啡馆（可能偶遇熟人）
4. 购物/办事
5. 继续工作/学习（如果你是尽责型性格）

请以JSON格式回复:
{
  "activity": "活动名称",
  "location": "地点",
  "reason": "选择这个活动的理由（简短）",
  "invite_friends": ["朋友ID1", "朋友ID2"] // 如果要邀请朋友，否则为空数组
}
    """ % [
        ai.character_name,
        day_type,
        personality_desc,
        mood,
        recent_memories,
        friends_str,
        int(time_slot.start_hour),
        int(time_slot.end_hour),
        time_slot.duration
    ]

    # 调用LLM
    var llm_response = await LLMService.call_llm(prompt, ai.id)
    var decision = JSON.parse_string(llm_response)

    if not decision:
        return null

    # 创建日程条目
    var entry = ScheduleEntry.new()
    entry.ai_id = ai.id
    entry.schedule_type = "dynamic"
    entry.activity_name = decision.get("activity", "自由活动")
    entry.location = decision.get("location", "家")
    entry.start_hour = time_slot.start_hour
    entry.end_hour = time_slot.end_hour
    entry.duration = time_slot.duration
    entry.participants = decision.get("invite_friends", [])
    entry.priority = 4
    entry.is_flexible = true
    entry.created_reason = decision.get("reason", "")
    entry.llm_generated = true

    # 如果要邀请朋友，发送邀约
    if entry.participants.size() > 0:
        _send_social_invitations(ai, entry)

    return entry

func _format_personality(p) -> String:

宜人性: %d/100 (%s)

神经质: %d/100 (%s)
开放性: %d/100 (%s)

        p.agreeableness, _get_trait_label(p.agreeableness),
        p.conscientiousness, _get_trait_label(p.conscientiousness),
        p.neuroticism, _get_trait_label(p.neuroticism),
        p.openness, _get_trait_label(p.openness)

    if value > 70: return "高"
```

## 🤝 社交邀约系统

### 邀约发起与响应

```gdscript
func _send_social_invitations(initiator: Node, activity: ScheduleEntry):
    """发送社交邀约给参与者"""

    for participant_id in activity.participants:
        var invitation = {
            "initiator_id": initiator.id,
            "activity": activity.activity_name,
            "location": activity.location,
            "start_hour": activity.start_hour,
            "duration": activity.duration,
            "invitation_time": TimeSystem.get_total_hours()
        }

        # 异步处理响应（给AI时间决策）
        _process_invitation_async(participant_id, invitation)

func _process_invitation_async(ai_id: String, invitation: Dictionary):
    """AI处理社交邀约"""

    var ai = _get_ai_node(ai_id)
    if not ai:
        return

    # 检查日程冲突
    var conflicts = _check_schedule_conflicts(ai, invitation.start_hour, invitation.duration)

    # 计算接受概率
    var accept_chance = _calculate_invitation_acceptance(ai, invitation, conflicts)

    if randf() < accept_chance:
        # 接受邀约
        _accept_invitation(ai, invitation, conflicts)
    else:
        # 拒绝邀约
        _decline_invitation(ai, invitation)

func _calculate_invitation_acceptance(ai: Node, invitation: Dictionary, conflicts: Array) -> float:
    """计算接受邀约的概率"""

    var base_chance = 0.6

    # 因素1: 性格
    var personality_modifier = 1.0
    personality_modifier += (ai.personality.extraversion - 50) / 100.0  # ±0.5
    personality_modifier += (ai.personality.agreeableness - 50) / 200.0  # ±0.25

    # 因素2: 关系
    var rel = RelationshipSystem.get_relationship(ai.id, invitation.initiator_id)
    var relationship_modifier = 1.0 + (rel.affection / 100.0)  # 0.0-2.0

    # 因素3: 心情
    var mood_modifier = 1.0
    if ai.current_mood < 30:
        mood_modifier = 0.5  # 心情差不想社交
    elif ai.current_mood > 70:
        mood_modifier = 1.2

    # 因素4: 日程冲突
    var conflict_penalty = 1.0
    if conflicts.size() > 0:
        var max_priority = 0
        for conflict in conflicts:
            max_priority = max(max_priority, conflict.priority)

        if max_priority >= 8:  # 高优先级冲突
            conflict_penalty = 0.2
        elif max_priority >= 5:
            conflict_penalty = 0.6
        else:
            conflict_penalty = 0.8

    # 因素5: 内向AI的社交疲劳
    if ai.personality.extraversion < 40:
        var social_count_today = _count_social_activities_today(ai)
        if social_count_today >= 2:
            mood_modifier *= 0.5  # 已经社交太多了

    var final_chance = base_chance * personality_modifier * relationship_modifier * mood_modifier * conflict_penalty

    return clamp(final_chance, 0.0, 1.0)

func _accept_invitation(ai: Node, invitation: Dictionary, conflicts: Array):
    """接受邀约"""

    # 取消冲突的低优先级活动
    for conflict in conflicts:
        if conflict.priority < 8 and conflict.is_flexible:
            conflict.is_canceled = true
            print("[日程] %s 取消了 %s 以参加 %s 的邀约" % [
                ai.character_name,
                conflict.activity_name,
                invitation.initiator_id
            ])

    # 添加新活动到日程
    var entry = ScheduleEntry.new()
    entry.ai_id = ai.id
    entry.schedule_type = "social"
    entry.activity_name = invitation.activity
    entry.location = invitation.location
    entry.start_hour = invitation.start_hour
    entry.duration = invitation.duration
    entry.participants = [invitation.initiator_id]
    entry.priority = 6

    _add_to_schedule(ai, entry)

    # 通知发起者
    _notify_invitation_accepted(invitation.initiator_id, ai.id)

    print("[社交] %s 接受了 %s 的邀约: %s" % [
        ai.character_name,
        invitation.initiator_id,
        invitation.activity
    ])

func _decline_invitation(ai: Node, invitation: Dictionary):
    """拒绝邀约"""

    # 通知发起者
    _notify_invitation_declined(invitation.initiator_id, ai.id)

    # 轻微影响关系（-2好感度）
    var rel = RelationshipSystem.get_relationship(invitation.initiator_id, ai.id)
    rel.affection = max(rel.affection - 2.0, -100.0)

    print("[社交] %s 拒绝了 %s 的邀约: %s" % [
        ai.character_name,
        invitation.initiator_id,
        invitation.activity
    ])
```

## ⏰ 日程执行与中断

### 实时日程跟踪

```gdscript
class ScheduleManager:
    """日程管理器（每个AI一个实例）"""

    var ai: Node
    var current_schedule: Array = []
    var current_activity: ScheduleEntry = null
    var activity_start_time: float = 0.0

    func _ready():
        # 连接时间系统信号
        TimeSystem.time_updated.connect(_on_time_updated)

    func _on_time_updated(current_hour: float):
        """每个时间单位（10分钟）检查日程"""

        # 检查是否需要开始新活动
        for entry in current_schedule:
            if entry.is_canceled:
                continue

            if abs(current_hour - entry.start_hour) < 0.1:  # 10分钟容差
                _start_activity(entry)
                break

        # 检查当前活动是否结束
        if current_activity:
            var elapsed = current_hour - activity_start_time
            if elapsed >= current_activity.duration:
                _end_activity()

    func _start_activity(entry: ScheduleEntry):
        """开始活动"""

        # 结束上一个活动
        if current_activity:
            _end_activity()

        current_activity = entry
        activity_start_time = TimeSystem.get_current_hour()

        # 移动AI到活动地点
        ai.move_to_location(entry.location)

        # 创建记忆
        MemorySystem.create_memory(
            ai.id,
            _memory_type_from_schedule(entry.schedule_type),
            "开始活动: " + entry.activity_name,
            entry.participants,
            entry.location,
            0.3,  # 低情感强度
            0.1   # 轻微正面
        )

        print("[日程] %s 开始活动: %s @ %s" % [
            ai.character_name,
            entry.activity_name,
            entry.location
        ])

        # 如果是社交活动，触发互动
        if entry.schedule_type == "social" and entry.participants.size() > 0:
            _trigger_social_interaction(entry)

    func _end_activity():
        """结束活动"""

        if not current_activity:
            return

        print("[日程] %s 结束活动: %s" % [
            ai.character_name,
            current_activity.activity_name
        ])

        # 根据活动类型应用效果
        _apply_activity_effects(current_activity)

        current_activity = null

    func _apply_activity_effects(entry: ScheduleEntry):
        """应用活动效果（心情、关系、压力）"""

        match entry.schedule_type:
            "work":
                ai.work_data.stress_level += 5
                if entry.activity_name == "overtime":
                    ai.work_data.stress_level += 10
                    ai.current_mood -= 5

            "social":
                ai.current_mood += 10
                if ai.personality.extraversion > 60:
                    ai.current_mood += 5  # 外向AI更享受社交

                # 增进关系
                for participant_id in entry.participants:
                    RelationshipSystem.add_interaction(ai.id, participant_id, 0.7)

            "personal_time":
                if ai.personality.extraversion < 40:
                    ai.current_mood += 10  # 内向AI需要独处
                ai.work_data.stress_level = max(ai.work_data.stress_level - 10, 0)

    func interrupt_activity(reason: String, new_activity: ScheduleEntry = null):
        """中断当前活动"""

        if not current_activity:
            return

        print("[日程中断] %s 的 %s 被中断: %s" % [
            ai.character_name,
            current_activity.activity_name,
            reason
        ])
```

```gdscript
func _calculate_free_time_slots(schedule: Array) -> Array:
    """计算空闲时间段"""
    var occupied_hours = []

    for entry in schedule:
        if not entry.is_canceled:
            occupied_hours.append({"start": entry.start_hour, "end": entry.end_hour})

    # 排序
    occupied_hours.sort_custom(func(a, b): return a.start < b.start)

    # 找空隙
    var free_slots = []
    var day_start = 6.0  # 早上6点开始
    var day_end = 24.0   # 晚上12点结束

    var current_time = day_start
    for occupied in occupied_hours:
        if occupied.start - current_time >= 1.0:  # 至少1小时空闲
            free_slots.append({
                "start_hour": current_time,
                "end_hour": occupied.start,
                "duration": occupied.start - current_time
            })
        current_time = max(current_time, occupied.end)

    # 最后一段时间
    if day_end - current_time >= 1.0:
        free_slots.append({
            "start_hour": current_time,
            "end_hour": day_end,
            "duration": day_end - current_time
        })

    return free_slots

func _resolve_schedule_conflicts(schedule: Array) -> Array:
    """解决日程冲突"""
    var resolved = []

    for entry in schedule:
        var has_conflict = false

        for existing in resolved:
            if _is_time_overlap(entry, existing):
                # 保留优先级高的
                if entry.priority > existing.priority:
                    existing.is_canceled = true
                else:
                    has_conflict = true
                break

        if not has_conflict:
            resolved.append(entry)

    return resolved.filter(func(e): return not e.is_canceled)

func _is_time_overlap(a: ScheduleEntry, b: ScheduleEntry) -> bool:
    """检查两个活动是否时间重叠"""
    return not (a.end_hour <= b.start_hour or a.start_hour >= b.end_hour)
```
