### 2.1 TaskSystem - 任务系统

#### 核心类结构

```gdscript
# script/task/TaskSystem.gd
extends Node

# ========================================
# 数据结构
# ========================================

var active_tasks: Dictionary = {}  # {ai_id: [task1, task2, ...]}
var task_history: Dictionary = {}  # 任务历史

# ========================================
# 信号
# ========================================

signal task_created(ai_id: String, task_id: String)
signal task_started(ai_id: String, task_id: String)
signal task_completed(ai_id: String, task_id: String, result: Dictionary)
signal task_failed(ai_id: String, task_id: String, reason: String)
signal social_invitation_sent(inviter_id: String, invitee_id: String, task: Dictionary)

# ========================================
# 初始化
# ========================================

func _ready():
    print("[TaskSystem] 任务系统初始化")

    # 连接时间系统
    TimeSystem.hour_changed.connect(_on_hour_changed)
    TimeSystem.day_changed.connect(_on_day_changed)

    # 连接天气系统
    WeatherSystem.weather_changed.connect(_on_weather_changed)

# ========================================
# 任务生成
# ========================================

func generate_daily_tasks(ai: Node):
    """为AI生成今日任务"""

    # 1. 生成工作任务
    if not TimeSystem.is_weekend() or ai.career in ["barista", "waiter"]:
        var work_tasks = generate_daily_work_tasks(ai)
        for task in work_tasks:
            add_task(ai.id, task)

    # 2. 生成日常任务
    var daily_tasks = generate_daily_routine_tasks(ai)
    for task in daily_tasks:
        add_task(ai.id, task)

func generate_daily_work_tasks(ai: Node) -> Array:
    """生成工作任务"""
    var career_id = ai.career
    var career_data = CareerSystem.get_career_data(career_id)
    var schedule_template = career_data.get("schedule_template", [])

    var tasks = []

    for schedule_item in schedule_template:
        var activity = schedule_item.get("activity", "")
        if activity in ["arrive_office", "leave_office", "lunch", "break"]:
            continue

        var task = _create_task_from_activity(ai, activity, schedule_item)
        if task:
            tasks.append(task)

    # 10%概率生成紧急任务
    if randf() < 0.1:
        var urgent_task = _generate_urgent_task(ai)
        if urgent_task:
            tasks.insert(0, urgent_task)

    return tasks

# ========================================
# 任务管理
# ========================================

func add_task(ai_id: String, task: Dictionary):
    if not active_tasks.has(ai_id):
        active_tasks[ai_id] = []

    active_tasks[ai_id].append(task)
    task_created.emit(ai_id, task.id)

func get_next_task(ai_id: String) -> Dictionary:
    """获取AI下一个应该做的任务"""
    if not active_tasks.has(ai_id):
        return {}

    var current_hour = TimeSystem.get_current_hour()
    var pending_tasks = []

    for task in active_tasks[ai_id]:
        if task.status == TaskData.TaskStatus.PENDING:
            if task.scheduled_time <= current_hour:
                pending_tasks.append(task)

    if pending_tasks.size() == 0:
        return {}

    # 按优先级排序
    pending_tasks.sort_custom(func(a, b): return a.priority < b.priority)

    return pending_tasks[0]

# ========================================
# 任务完成
# ========================================

func complete_task(ai: Node, task: Dictionary) -> Dictionary:
    """完成任务"""
    task.status = TaskData.TaskStatus.IN_REVIEW
    task.completed_at = TimeSystem.get_current_timestamp()
    task.actual_duration = task.completed_at - task.started_at

    var result = {}

    # 根据任务复杂度选择判定方式
    if task.difficulty <= TaskData.TaskDifficulty.LOW:
        result = _evaluate_simple_task(ai, task)
    else:
        result = await _evaluate_complex_task_with_llm(ai, task)

    # 应用奖励/惩罚
    if result.requirements_met:
        _apply_task_rewards(ai, task, result)
        task.status = TaskData.TaskStatus.COMPLETED
    else:
        _apply_task_penalties(ai, task, result)
        task.status = TaskData.TaskStatus.FAILED

    # 创建记忆
    _create_task_memory(ai, task, result)

    # 更新绩效
    if ai.has_property("work_data"):
        ai.work_data.performance_score += result.performance_impact

    # 发出信号
    task_completed.emit(ai.id, task.id, result)

    return result

# ========================================
# 事件响应
# ========================================

func _on_hour_changed(new_hour: int):
    """每小时检查任务"""
    for ai_id in active_tasks.keys():
        var ai = _get_ai_by_id(ai_id)
        if not ai:
            continue

        var next_task = get_next_task(ai_id)
        if next_task:
            ai.on_new_task_available(next_task)

func _on_day_changed(year: int, season: int, day: int):
    """每天生成新任务"""
    var all_ais = CharacterManager.get_all_characters()
    for ai in all_ais:
        generate_daily_tasks(ai)

func _on_weather_changed(new_weather: String, old_weather: String):
    """天气变化影响任务"""
    # 大雨/雷暴时取消户外任务
    if new_weather in ["heavy_rain", "thunderstorm"]:
        for ai_id in active_tasks.keys():
            for task in active_tasks[ai_id]:
                if task.get("is_outdoor", false) and task.status == TaskData.TaskStatus.PENDING:
                    task.status = TaskData.TaskStatus.CANCELED
                    print("[TaskSystem] %s 的户外任务因恶劣天气取消: %s" % [ai_id, task.title])
```

#### 核心类结构

```gdscript
# script/weather/WeatherSystem.gd
extends Node

# ========================================
# 常量
# ========================================

const WeatherType = {
    SUNNY = "sunny",
    CLOUDY = "cloudy",
    LIGHT_RAIN = "light_rain",
    HEAVY_RAIN = "heavy_rain",
    THUNDERSTORM = "thunderstorm",
    SNOWY = "snowy",
    FOGGY = "foggy",
    WINDY = "windy"
}

const WEATHER_UPDATE_INTERVAL = 6.0  # 每6游戏小时检查一次

# ========================================
# 状态
# ========================================

var current_weather: String = WeatherType.SUNNY
var weather_intensity: float = 1.0
var weather_duration_remaining: float = 6.0
var last_weather_check_hour: float = 0.0

# ========================================
# 信号
# ========================================

signal weather_changed(new_weather: String, old_weather: String)
signal weather_intensity_changed(intensity: float)

# ========================================
# 初始化
# ========================================

func _ready():
    print("[WeatherSystem] 天气系统初始化")

    # 连接时间系统
    TimeSystem.hour_changed.connect(_on_hour_changed)
    TimeSystem.season_changed.connect(_on_season_changed)

    # 根据季节初始化天气
    _generate_initial_weather()

func _generate_initial_weather():
    var season = TimeSystem.current_season
    current_weather = _pick_weather_by_season(season)
    weather_duration_remaining = randf_range(4.0, 10.0)

# ========================================
# 天气更新
# ========================================

func _on_hour_changed(new_hour: int):
    var current_hour = TimeSystem.get_current_hour()

    weather_duration_remaining -= 1.0

    if weather_duration_remaining <= 0 or (current_hour - last_weather_check_hour) >= WEATHER_UPDATE_INTERVAL:
        _update_weather()
        last_weather_check_hour = current_hour

func _update_weather():
    var old_weather = current_weather
    var season = TimeSystem.current_season

    var new_weather = _generate_next_weather(old_weather, season)

    if new_weather != old_weather:
        current_weather = new_weather
        weather_duration_remaining = randf_range(3.0, 12.0)
        weather_changed.emit(new_weather, old_weather)

        print("[WeatherSystem] 天气变化: %s → %s" % [
            _get_weather_name(old_weather),
            _get_weather_name(new_weather)
        ])

        _apply_weather_effects()

func _generate_next_weather(current: String, season: String) -> String:
    """生成下一个天气"""

    # 冬季特殊处理
    if season == "Winter" and randf() < 0.6:
        return _pick_winter_weather()

    # 根据当前天气的转换概率矩阵
    var transition_matrix = {
        "sunny": {"sunny": 0.6, "cloudy": 0.3, "light_rain": 0.08, "windy": 0.02},
        "cloudy": {"cloudy": 0.4, "sunny": 0.3, "light_rain": 0.2, "heavy_rain": 0.05, "thunderstorm": 0.03, "foggy": 0.02},
        "light_rain": {"light_rain": 0.4, "cloudy": 0.3, "heavy_rain": 0.2, "thunderstorm": 0.05, "sunny": 0.05},
        "heavy_rain": {"heavy_rain": 0.5, "thunderstorm": 0.2, "light_rain": 0.2, "cloudy": 0.1},
        "thunderstorm": {"thunderstorm": 0.3, "heavy_rain": 0.5, "light_rain": 0.15, "cloudy": 0.05},
        "snowy": {"snowy": 0.6, "cloudy": 0.3, "sunny": 0.1},
        "foggy": {"foggy": 0.4, "cloudy": 0.4, "sunny": 0.2},
        "windy": {"windy": 0.3, "sunny": 0.4, "cloudy": 0.3}
    }

    var transitions = transition_matrix.get(current, {"sunny": 1.0})
    transitions = _apply_season_modifiers(transitions, season)

    return _weighted_random_choice(transitions)

# ========================================
# 天气效果应用
# ========================================

func _apply_weather_effects():
    var weather_data = get_current_weather_data()

    # 更新环境光照
    _update_ambient_light(weather_data)

    # 更新天空颜色
    _update_sky_color(weather_data)

    # 更新粒子效果
    _update_weather_particles(weather_data)

    # 更新音效
    _update_weather_sound(weather_data)

    # 通知AI系统天气变化
    _notify_ai_weather_change()

# ========================================
# 查询API
# ========================================

func get_current_weather() -> String:
    return current_weather

func get_ai_mood_modifier() -> int:
    var data = WEATHER_DATA.get(current_weather, {})
    return data.get("ai_mood_modifier", 0)

func get_outdoor_activity_modifier() -> float:
    var data = WEATHER_DATA.get(current_weather, {})
    return data.get("outdoor_activity_modifier", 0.0)

func should_force_indoor() -> bool:
    var data = WEATHER_DATA.get(current_weather, {})
    return data.get("force_indoor", false)

func is_raining() -> bool:
    return current_weather in [WeatherType.LIGHT_RAIN, WeatherType.HEAVY_RAIN, WeatherType.THUNDERSTORM]

func is_snowing() -> bool:
    return current_weather == WeatherType.SNOWY
```

    var interaction_points: Dictionary = {}
    var description: String = ""

    func to_dict() -> Dictionary:
        return {
            "id": id,
            "name_cn": name_cn,
            "name_en": name_en,
            "position": {"x": position.x, "y": position.y},
            "type": type,
            "capacity": capacity,
            "current_occupants": current_occupants.duplicate(),
            "interaction_points": interaction_points.duplicate()
        }

# ========================================
# 地点类型枚举
# ========================================

enum LocationType {
    OFFICE,
    COFFEE_SHOP,
    HOME,
    MEETING_ROOM,
    BATHROOM,
    CAFETERIA,
    GYM,
    PARK,
    LIBRARY,
    ENTRANCE,
    HALLWAY,
    LOUNGE,
    OTHER
}

# ========================================
# 数据
# ========================================

var locations: Dictionary = {}  # {location_id: Location}

# ========================================
# 信号
# ========================================

signal location_registered(location_id: String)
signal location_occupant_changed(location_id: String, occupants_count: int)

# ========================================
# 初始化
# ========================================

func _ready():
    print("[LocationManager] 地点管理系统初始化")

    # 等待一帧后加载场景中的地点
    await get_tree().process_frame
    _load_locations_from_scene()

func _load_locations_from_scene():
    """从场景加载地点标记"""
    var location_nodes = get_tree().get_nodes_in_group("location")

    for node in location_nodes:
        if node.has_meta("location_id"):
            var id = node.get_meta("location_id")
            var name_cn = node.get_meta("location_name_cn", id)
            var type_str = node.get_meta("location_type", "OTHER")
            var type = _parse_location_type(type_str)
            var capacity = node.get_meta("location_capacity", 10)

            register_location(id, name_cn, node.global_position, type, capacity)

    print("  - 从场景加载了 %d 个地点" % locations.size())

# ========================================
# 地点注册
# ========================================

func register_location(
    id: String,
    name_cn: String,
    position: Vector2,
    type: LocationType,
    capacity: int = 10,
    interaction_points: Dictionary = {}
) -> Location:
    """注册一个新地点"""

    var location = Location.new()
    location.id = id
    location.name_cn = name_cn
    location.name_en = _translate_to_english(name_cn)
    location.position = position
    location.type = type
    location.capacity = capacity
    location.interaction_points = interaction_points

    locations[id] = location
    location_registered.emit(id)

    print("[LocationManager] 注册地点: %s (%s)" % [name_cn, id])

    return location

# ========================================
# 地点查询
# ========================================

func get_location(id: String) -> Location:
    return locations.get(id, null)

func get_location_by_name(name_cn: String) -> Location:
    for location in locations.values():
        if location.name_cn == name_cn:
            return location
    return null

func get_locations_by_type(type: LocationType) -> Array:
    var result = []
    for location in locations.values():
        if location.type == type:
            result.append(location)
    return result

func get_nearest_location(from_position: Vector2, type: LocationType = -1) -> Location:
    """获取最近的地点"""
    var nearest: Location = null
    var min_distance = INF

    for location in locations.values():
        if type != -1 and location.type != type:
            continue

        var distance = from_position.distance_to(location.position)
        if distance < min_distance:
            min_distance = distance
            nearest = location

    return nearest

func get_available_location(type: LocationType) -> Location:
    """获取指定类型的未满地点"""
    var available_locations = []

    for location in get_locations_by_type(type):
        if location.current_occupants.size() < location.capacity:
            available_locations.append(location)

    if available_locations.size() == 0:
        return null

    # 随机返回一个
    return available_locations.pick_random()

# ========================================
# 占用管理
# ========================================

func set_location_occupant(location_id: String, ai_id: String, entering: bool):
    """设置AI进入/离开地点"""
    var location = get_location(location_id)
    if not location:
        return

    if entering:
        if not location.current_occupants.has(ai_id):
            location.current_occupants.append(ai_id)
    else:
        location.current_occupants.erase(ai_id)

    location_occupant_changed.emit(location_id, location.current_occupants.size())

func get_occupants(location_id: String) -> Array:
    var location = get_location(location_id)
    if not location:
        return []
    return location.current_occupants.duplicate()
```

# ========================================

enum ConflictType {
    IDEOLOGICAL,     # 意识形态冲突
    INTEREST,        # 利益冲突

enum SeverityLevel {
    MINOR = 1,       # 小矛盾
    MODERATE = 2,    # 中等矛盾
    SERIOUS = 3,     # 严重矛盾
    MAJOR = 4,       # 重大矛盾
    CRITICAL = 5     # 危机级矛盾
}

# ========================================
# 数据
# ========================================

var active_conflicts: Dictionary = {}  # {conflict_id: ConflictData}
var conflict_history: Array = []

# ========================================
# 信号
# ========================================

func _ready():

    # 连接关系系统
    RelationshipManager.ai_relationship_changed.connect(_on_relationship_changed)

    # 连接任务系统
    TaskSystem.task_failed.connect(_on_task_failed)

    # 每小时检查冲突

# ========================================
# 冲突触发
# ========================================

func check_conflict_probability(ai_id: String, target_id: String, context: Dictionary) -> float:
    """计算冲突概率"""

    var base_probability = 0.05  # 基础5%

    # 性格因素
    var ai_personality = PersonalityEngine.get_personality(ai_id)
    var target_personality = PersonalityEngine.get_personality(target_id)

    var personality_factor = 1.0
    # 神经质高的AI更容易冲突
    personality_factor *= (ai_personality.neuroticism / 50.0)
    # 宜人性低的AI更容易冲突
    personality_factor *= (2.0 - target_personality.agreeableness / 50.0)

    # 关系因素
    var relationship = RelationshipManager.get_relationship(ai_id, target_id)
    var relationship_factor = 1.0
    if relationship.affection < 30:
        relationship_factor = 2.0  # 关系差,概率×2
    elif relationship.affection > 70:
        relationship_factor = 0.5  # 关系好,概率×0.5

    # 情绪因素
    var emotional_factor = 1.0
    if context.has("ai_stress") and context.ai_stress > 70:
        emotional_factor *= 1.5
    if context.has("ai_anger") and context.ai_anger > 50:
        emotional_factor *= 2.0

    # 环境因素
    var environment_factor = 1.0
    if WeatherSystem.current_weather in ["thunderstorm", "heavy_rain"]:
        environment_factor *= 1.2  # 坏天气增加冲突概率

    var final_probability = base_probability * personality_factor * relationship_factor * emotional_factor * environment_factor

    return clamp(final_probability, 0.0, 0.5)  # 最高50%

func trigger_conflict(ai_id: String, target_id: String, conflict_type: ConflictType, context: Dictionary):
    """触发冲突"""

    var conflict_id = _generate_conflict_id()

    var conflict_data = {
        "id": conflict_id,
        "type": conflict_type,
        "involved_characters": [ai_id, target_id],
        "mediators": [],
        "severity_level": SeverityLevel.MINOR,
        "status": "ACTIVE",
        "trigger_event": context,
        "trigger_timestamp": TimeSystem.get_current_timestamp(),
        "escalation_history": [],
        "resolution": null
    }

    active_conflicts[conflict_id] = conflict_data

    # 创建记忆
    MemoryManager.add_memory(
        ai_id,
        "我和%s发生了冲突,因为%s" % [target_id, context.get("reason", "")],
        MemoryManager.MemoryType.INTERACTION,
        MemoryManager.MemoryImportance.HIGH
    )

    # 降低关系
    RelationshipManager.update_relationship(ai_id, target_id, "affection", -10)
    RelationshipManager.update_relationship(ai_id, target_id, "trust", -5)

    conflict_triggered.emit(conflict_id, [ai_id, target_id])

    print("[ConflictSystem] 冲突触发: %s vs %s (类型:%d)" % [ai_id, target_id, conflict_type])

# ========================================
# 冲突升级
# ========================================

func check_conflict_escalation(conflict_id: String):
    """检查冲突是否升级"""
    var conflict = active_conflicts.get(conflict_id)
    if not conflict:
        return

    # 升级条件
    var should_escalate = false
    var escalation_probability = 0.2  # 基础20%

    # 已经是最高级,不再升级
    if conflict.severity_level >= SeverityLevel.CRITICAL:
        return

    # 时间因素:冲突持续时间越长,越容易升级
    var duration = TimeSystem.get_current_timestamp() - conflict.trigger_timestamp
    if duration > 3600:  # 持续超过1小时
        escalation_probability += 0.2

    # 第三方介入因素
    if conflict.mediators.size() == 0:
        escalation_probability += 0.1

    if randf() < escalation_probability:
        _escalate_conflict(conflict_id)

func _escalate_conflict(conflict_id: String):
    """升级冲突"""
    var conflict = active_conflicts.get(conflict_id)
    if not conflict:
        return

    var old_severity = conflict.severity_level
    conflict.severity_level += 1

    conflict.escalation_history.append({
        "timestamp": TimeSystem.get_current_timestamp(),
        "severity": conflict.severity_level,
        "description": "冲突升级"
    })

    # 进一步降低关系
    var ai_id = conflict.involved_characters[0]
    var target_id = conflict.involved_characters[1]
    RelationshipManager.update_relationship(ai_id, target_id, "affection", -5)

    conflict_escalated.emit(conflict_id, conflict.severity_level)

    print("[ConflictSystem] 冲突升级: %s (严重度: %d -> %d)" % [conflict_id, old_severity, conflict.severity_level])

# ========================================
# 冲突解决
# ========================================

func attempt_resolution(conflict_id: String, mediator_id: String = ""):
    """尝试解决冲突"""
    var conflict = active_conflicts.get(conflict_id)
    if not conflict:
        return

    var resolution_probability = 0.5  # 基础50%

    # 有调解者提高概率
    if mediator_id != "":
        conflict.mediators.append(mediator_id)
        var mediator_personality = PersonalityEngine.get_personality(mediator_id)
        # 宜人性高的调解者更有效
        resolution_probability += mediator_personality.agreeableness / 200.0  # +0~0.5

    # 严重度越高越难解决
    resolution_probability -= (conflict.severity_level - 1) * 0.1

    if randf() < resolution_probability:
        _resolve_conflict(conflict_id, "MEDIATION" if mediator_id != "" else "COMPROMISE")
    else:
        print("[ConflictSystem] 冲突解决失败: %s" % conflict_id)

func _resolve_conflict(conflict_id: String, resolution_type: String):
    """解决冲突"""
    var conflict = active_conflicts.get(conflict_id)
    if not conflict:
        return

    conflict.status = "RESOLVED"
    conflict.resolution = {
        "resolution_type": resolution_type,
        "resolution_timestamp": TimeSystem.get_current_timestamp(),
        "mediator": conflict.mediators[0] if conflict.mediators.size() > 0 else "",
        "outcome": "达成妥协"
    }

    # 轻微恢复关系
    var ai_id = conflict.involved_characters[0]
    var target_id = conflict.involved_characters[1]
    RelationshipManager.update_relationship(ai_id, target_id, "affection", 3)

    # 创建记忆
    MemoryManager.add_memory(
        ai_id,
        "我和%s的冲突得到了解决" % target_id,
        MemoryManager.MemoryType.INTERACTION,
        MemoryManager.MemoryImportance.NORMAL
    )

    # 移到历史记录
    conflict_history.append(conflict)
    active_conflicts.erase(conflict_id)

    conflict_resolved.emit(conflict_id, resolution_type)

    print("[ConflictSystem] 冲突已解决: %s (方式:%s)" % [conflict_id, resolution_type])

# ========================================
# 事件响应
# ========================================

func _on_relationship_changed(ai_id: String, target_id: String, relationship: Dictionary):
    """关系变化时检查冲突"""
    # 如果关系降到很低,可能触发冲突
    if relationship.affection < 20 and relationship.trust < 30:
        var conflict_probability = check_conflict_probability(ai_id, target_id, {
            "reason": "关系恶化",
            "ai_stress": 50
        })

        if randf() < conflict_probability:
            trigger_conflict(ai_id, target_id, ConflictType.EMOTIONAL, {
                "reason": "长期积累的矛盾爆发"
            })

func _on_task_failed(ai_id: String, task_id: String, reason: String):
    """任务失败时可能触发冲突"""
    var task = TaskSystem.get_task(ai_id, task_id)
    if not task:
        return

    # 如果任务涉及其他AI,可能引发冲突
    if task.has("related_ai") and task.related_ai.size() > 0:
        for other_ai in task.related_ai:
            var conflict_probability = check_conflict_probability(ai_id, other_ai, {
                "reason": "任务失败导致责任归属争议",
                "ai_stress": 70
            })

            if randf() < conflict_probability:
                trigger_conflict(ai_id, other_ai, ConflictType.INTEREST, {
                    "reason": "因任务失败产生利益冲突"
                })

func _on_hour_changed(hour: int):
    """每小时检查活跃冲突"""
    for conflict_id in active_conflicts.keys():
        check_conflict_escalation(conflict_id)

        # 10%概率自然解决(如果是MINOR级别)
        var conflict = active_conflicts[conflict_id]
        if conflict.severity_level == SeverityLevel.MINOR and randf() < 0.1:
            _resolve_conflict(conflict_id, "NATURAL")

    var nearest_location = LocationManager.get_nearest_location(

        location_type
    )

    if not nearest_location:
        movement_failed.emit("未找到类型为%d的地点" % location_type)
        return false

    return move_to_location(nearest_location.id, callback)

func execute_schedule_movement(activity_type: String, location_id: String = ""):
    """根据日程活动移动"""
    if location_id != "":
        move_to_location(location_id)
        return

    # 根据活动类型自动选择地点
    match activity_type:
        "work":
            move_to_nearest_location(LocationManager.LocationType.OFFICE)
        "meeting":
            move_to_nearest_location(LocationManager.LocationType.MEETING_ROOM)
        "lunch", "dinner":
            move_to_nearest_location(LocationManager.LocationType.CAFETERIA)
        "coffee_break":
            move_to_nearest_location(LocationManager.LocationType.COFFEE_SHOP)
        "gym":
            move_to_nearest_location(LocationManager.LocationType.GYM)
        _:
            wander_around(150.0)

func wander_around(radius: float = 200.0):
    """随机游荡"""
    var random_offset = Vector2(
        randf_range(-radius, radius),
        randf_range(-radius, radius)
    )
    var target_position = character_controller.global_position + random_offset
    character_controller.move_to(target_position)
    current_state = MovementState.MOVING_TO_POSITION

func _on_navigation_completed():
    """导航完成回调"""
    if current_state == MovementState.MOVING_TO_LOCATION:
        # 到达地点
        LocationManager.set_location_occupant(target_location_id, character_controller.name, true)
        arrived_at_location.emit(target_location_id)

    current_state = MovementState.IDLE
    movement_completed.emit()

    if move_callback.is_valid():

func _on_task_created(ai_id: String, task_id: String):

    if ai_id != ai_character.name:
        return

    # 获取任务
    var task = TaskSystem.get_task(ai_id, task_id)
    if not task:
        return

    # 检查是否应该立即执行
    var current_hour = TimeSystem.get_current_hour()
    if task.scheduled_time <= current_hour:

func execute_task(task: Dictionary):
    """执行任务"""
    current_task = task
    task_progress = 0.0

    # 更新任务状态
    TaskSystem.start_task(ai_character.name, task.id)
    task_execution_started.emit(task.id)

    # 根据任务类型执行不同逻辑
    match task.type:
        "work":
            _execute_work_task(task)
        "daily":
            _execute_daily_task(task)
        "social":
            _execute_social_task(task)
        "growth":
            _execute_growth_task(task)

    """执行工作任务"""
    # 移动到工作地点
    var ai_movement = ai_character.get_node_or_null("Components/AIMovementController")
    if ai_movement and task.has("location"):
        ai_movement.move_to_location(task.location, func():
            _work_task_arrived_at_location(task)
        )
    else:
        _work_task_arrived_at_location(task)

func _work_task_arrived_at_location(task: Dictionary):
    """到达工作地点后"""
    print("[TaskExecutor] %s 开始工作任务: %s" % [ai_character.name, task.title])

    # 设置AI状态为工作中
    var ai_brain = ai_character.get_node_or_null("Components/AIBrain")
    if ai_brain:
        ai_brain.set_state("WORKING")

    # 模拟任务执行 (实际游戏中会有更复杂的逻辑)
    var task_duration = task.duration * 3600  # 转换为秒
    await get_tree().create_timer(task_duration).timeout

    _complete_current_task()

    """完成当前任务"""
    if current_task.is_empty():

    task_progress = 1.0
    task_execution_progress.emit(current_task.id, 1.0)

# script/conflict/ConflictHandler.gd
extends Node

var ai_character: Node
var involved_conflicts: Array = []  # 参与的冲突ID列表

signal conflict_encountered(conflict_id: String)
signal conflict_resolved_by_self(conflict_id: String)

func _ready():
    ai_character = get_parent()

    # 连接冲突系统
    ConflictSystem.conflict_triggered.connect(_on_conflict_triggered)
    ConflictSystem.conflict_resolved.connect(_on_conflict_resolved)

func _on_conflict_triggered(conflict_id: String, involved_characters: Array):
    """冲突触发时"""
    if ai_character.name in involved_characters:
        involved_conflicts.append(conflict_id)
        conflict_encountered.emit(conflict_id)

        # 决定如何应对冲突
        _decide_conflict_response(conflict_id)

func _decide_conflict_response(conflict_id: String):
    """决定如何应对冲突"""
    var conflict = ConflictSystem.active_conflicts.get(conflict_id)
    if not conflict:
        return

    # 根据性格决定
    var personality = PersonalityEngine.get_personality(ai_character.name)

    # 宜人性高的AI更可能选择妥协/和解
    if personality.agreeableness > 70:
        # 尝试主动和解
        _attempt_peaceful_resolution(conflict_id)
    # 神经质高的AI更可能选择逃避
    elif personality.neuroticism > 70:
        # 逃避冲突
        _avoid_conflict(conflict_id)
    # 外向性高的AI更可能选择直接对抗
    elif personality.extraversion > 70:
        # 直接对抗
        _confront_conflict(conflict_id)
    else:
        # 寻求第三方调解
        _seek_mediation(conflict_id)

func _attempt_peaceful_resolution(conflict_id: String):
    """尝试和平解决"""
    print("[ConflictHandler] %s 尝试和平解决冲突 %s" % [ai_character.name, conflict_id])

    # 调用LLM生成和解对话
    var ai_brain = ai_character.get_node_or_null("Components/AIBrain")
    if ai_brain:
        ai_brain.generate_conflict_resolution_dialogue(conflict_id)

    # 尝试解决冲突
    ConflictSystem.attempt_resolution(conflict_id)

func _avoid_conflict(conflict_id: String):
    """逃避冲突"""
    print("[ConflictHandler] %s 选择逃避冲突 %s" % [ai_character.name, conflict_id])

    # 远离冲突对象
    var conflict = ConflictSystem.active_conflicts.get(conflict_id)
    var other_character_id = conflict.involved_characters[1] if conflict.involved_characters[0] == ai_character.name else conflict.involved_characters[0]

    var other_character = CharacterManager.get_character(other_character_id)
    if other_character:
        # 移动到远离对方的地点
        var ai_movement = ai_character.get_node_or_null("Components/AIMovementController")
        if ai_movement:
            ai_movement.wander_around(300.0)

func _seek_mediation(conflict_id: String):
    """寻求调解"""
    print("[ConflictHandler] %s 寻求第三方调解冲突 %s" % [ai_character.name, conflict_id])

    # 找一个关系好的朋友作为调解者
    var friends = RelationshipManager.get_friends(ai_character.name, "friend")
    if friends.size() > 0:
        var mediator_id = friends[0]
        ConflictSystem.attempt_resolution(conflict_id, mediator_id)

func _on_conflict_resolved(conflict_id: String, resolution_type: String):
    """冲突解决时"""
    if conflict_id in involved_conflicts:
        involved_conflicts.erase(conflict_id)
        conflict_resolved_by_self.emit(conflict_id)

        print("[ConflictHandler] %s 的冲突 %s 已解决 (方式:%s)" % [ai_character.name, conflict_id, resolution_type])

## 六、系统间集成流程示例

### 6.1 完整的AI日常行为流程

```
1. 时间系统触发新的一天
   TimeSystem.day_changed → TaskSystem

2. 任务系统生成今日任务
   TaskSystem.generate_daily_tasks(ai)
   - 生成工作任务 (基于CareerSystem)
   - 生成日常任务 (吃饭、睡觉)
   - 检查社交邀请

3. 天气系统影响任务
   WeatherSystem.weather_changed → TaskSystem
   - 大雨取消户外任务
   - 晴天增加户外活动

4. AI执行任务
   TaskExecutor.execute_task(task)
   - 使用AIMovementController移动到任务地点
   - LocationManager更新地点占用
   - 执行任务逻辑

5. 任务完成评估
   TaskSystem.complete_task(ai, task)
   - LLM评估任务质量 (如果是复杂任务)
   - 更新绩效分数 (EconomyManager)
   - 创建任务记忆 (MemoryManager)

6. 可能触发冲突
   如果任务失败或与他人产生矛盾:
   TaskSystem.task_failed → ConflictSystem
   - ConflictSystem计算冲突概率
   - 可能触发冲突
   - ConflictHandler决定应对方式

7. 关系系统更新
   ConflictSystem → RelationshipManager
   - 冲突降低关系
   - 或和解提升关系

8. 存档系统持久化
   SaveManager自动存档 (每天)
   - 收集所有系统数据
   - 序列化为JSON
   - 创建截图和备份

func _cmd_task_list(args: Array):
    if args.size() < 1:

    var ai_id = args[0]
    var tasks = TaskSystem.get_character_tasks(ai_id)

    var result = "AI %s 的任务列表:\n" % ai_id
    for task in tasks:
