func _upgrade_save_version(save_data: Dictionary, from_version: String, to_version: String) -> Dictionary:

	var upgraded_data = save_data.duplicate(true)

	# 版本升级链路
	match from_version:
		"1.0.0":
			if to_version >= "1.1.0":
				upgraded_data = _migrate_1_0_to_1_1(upgraded_data)
			if to_version >= "1.2.0":
				upgraded_data = _migrate_1_1_to_1_2(upgraded_data)
		"1.1.0":
			if to_version >= "1.2.0":
				upgraded_data = _migrate_1_1_to_1_2(upgraded_data)

	upgraded_data["meta"]["version"] = to_version
	return upgraded_data

**影响范围**:

**建议修复方案**:
```gdscript
func _calculate_importance_decay(perception: Dictionary) -> float:
	var base_importance = perception.get("importance", 5.0)
	var perception_time = perception.get("timestamp", 0)

	# 使用TimeSystem获取精确时间差
	var time_system = get_node_or_null("/root/TimeSystem")
	var time_passed_hours = 0.0

	if time_system:
		var current_time = time_system.get_current_timestamp()
		time_passed_hours = (current_time - perception_time) / 3600.0  # 秒转小时
	else:
		# 回退方案：使用系统时钟
		var current_ms = OS.get_ticks_msec()
		time_passed_hours = (current_ms - perception_time) / 3600000.0

	# 衰减曲线：每小时衰减10%，最多衰减50%
	var max_decay_hours = 5.0
	var decay_rate = 0.1  # 每小时10%
	var decay_factor = 1.0 - min(time_passed_hours * decay_rate, 0.5)

	return clamp(base_importance * decay_factor, 0.0, 10.0)
```
