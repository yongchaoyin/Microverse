## 👁️ 观察系统

### 系统概述

玩家作为观察者,可以随意观察AI角色的生活、工作、社交,但不能直接控制AI的行为。

### 观察工具

```gdscript
# ObservationSystem.gd
class_name ObservationSystem extends Node

# 观察模式
enum ObservationMode {
    FREE_CAMERA,        # 自由相机漫游
    FOLLOW_AI,          # 跟随特定AI
    OVERVIEW,           # 俯瞰全镇
    DATA_OVERLAY        # 数据叠加层
}

var current_mode: ObservationMode = ObservationMode.FREE_CAMERA
var followed_ai: Node = null
var camera: Camera2D

# 切换观察模式
func switch_observation_mode(mode: ObservationMode):
    current_mode = mode

    match mode:
        ObservationMode.FREE_CAMERA:
            enable_free_camera()
        ObservationMode.FOLLOW_AI:
            if followed_ai:
                camera.position = followed_ai.global_position
        ObservationMode.OVERVIEW:
            set_camera_overview()
        ObservationMode.DATA_OVERLAY:
            show_data_overlay()

# 选择AI进行跟随观察
func follow_ai(ai_character: Node):
    followed_ai = ai_character
    current_mode = ObservationMode.FOLLOW_AI
    camera.smooth_follow_target(ai_character)

    # 显示AI的实时数据面板
    show_ai_info_panel(ai_character)

# 显示AI信息面板
func show_ai_info_panel(ai: Node):
    var panel_data = {
        "name": ai.name,
        "current_activity": ai.current_activity,
        "location": ai.current_location,
        "mood": ai.mood,
        "energy": ai.energy,
        "current_thoughts": ai.get_current_thoughts(),
        "nearby_ai": ai.get_nearby_characters(),
        "relationships": ai.get_top_relationships()
    }

    UI.show_observation_panel(panel_data)
```

```gdscript
# SocialNetworkVisualizer.gd
class_name SocialNetworkVisualizer extends Control

# 绘制关系图谱
func draw_relationship_graph():
    var all_ai = get_tree().get_nodes_in_group("character")

    # 使用力导向图布局
    var graph_nodes = []
    var graph_edges = []

    for ai in all_ai:
        # 添加节点
        var node = create_graph_node(ai)
        graph_nodes.append(node)

        # 添加边(关系线)
        for relationship_target in ai.relationships:
            var edge = create_graph_edge(ai, relationship_target)
            graph_edges.append(edge)

    # 应用力导向布局算法
    apply_force_directed_layout(graph_nodes, graph_edges)

    # 渲染
    render_graph(graph_nodes, graph_edges)

# 创建图节点
func create_graph_node(ai: Node) -> Dictionary:
    return {
        "ai": ai,
        "position": Vector2.ZERO,  # 初始位置
        "color": get_ai_mood_color(ai),  # 根据心情着色
        "size": 20 + ai.social_influence * 10  # 社交影响力越大节点越大
    }

# 创建关系边
func create_graph_edge(ai_a: Node, ai_b: String) -> Dictionary:
    var ai_b_node = find_ai_by_name(ai_b)
    var relationship_data = ai_a.relationships[ai_b]

    return {
        "from": ai_a,
        "to": ai_b_node,
        "strength": relationship_data.get("friendship", 0),
        "romance": relationship_data.get("romance", 0),
        "color": get_relationship_color(relationship_data),
        "thickness": relationship_data.get("friendship", 0) / 50.0  # 友谊度影响线条粗细
    }

# 根据关系类型返回颜色
func get_relationship_color(relationship: Dictionary) -> Color:
    if relationship.get("romance", 0) > 50:
        return Color.PINK  # 恋爱关系
    elif relationship.get("friendship", 0) > 70:
        return Color.GREEN  # 好友关系
    elif relationship.get("friendship", 0) < 30:
        return Color.GRAY  # 普通关系
    elif relationship.get("rivalry", 0) > 50:
        return Color.RED  # 敌对关系
    else:
        return Color.WHITE
```

## 📊 数据分析系统

### AI行为统计

```gdscript
# AIAnalyticsSystem.gd
class_name AIAnalyticsSystem extends Node

# 数据收集
var ai_activity_log: Array = []
var daily_statistics: Dictionary = {}

# 记录AI活动
func log_ai_activity(ai: Node, activity_type: String, details: Dictionary):
    var log_entry = {
        "timestamp": Time.get_unix_time_from_system(),
        "ai_name": ai.name,
        "activity": activity_type,
        "details": details,
        "location": ai.global_position
    }

    ai_activity_log.append(log_entry)
    update_statistics(log_entry)

# 更新统计数据
func update_statistics(log_entry: Dictionary):
    var ai_name = log_entry["ai_name"]

    if not daily_statistics.has(ai_name):
        daily_statistics[ai_name] = {
            "work_hours": 0,
            "social_hours": 0,
            "rest_hours": 0,
            "conversations": 0,
            "places_visited": [],
            "money_spent": 0,
            "mood_average": 0
        }

    var stats = daily_statistics[ai_name]

    match log_entry["activity"]:
        "work":
            stats["work_hours"] += 1
        "social":
            stats["social_hours"] += 1
            stats["conversations"] += 1
        "rest":
            stats["rest_hours"] += 1
        "purchase":
            stats["money_spent"] += log_entry["details"].get("amount", 0)

# 生成每日报告
func generate_daily_report() -> Dictionary:
    var report = {
        "date": TimeSystem.get_current_date_string(),
        "ai_statistics": daily_statistics,
        "town_summary": {
            "total_conversations": count_total_conversations(),
            "total_money_spent": count_total_money_spent(),
            "most_active_ai": find_most_active_ai(),
            "most_popular_location": find_most_popular_location(),
            "relationship_changes": get_relationship_changes()
        }
    }

    return report

# 显示报告UI
func show_daily_report_ui():
    var report = generate_daily_report()
    var report_ui = preload("res://scene/ui/DailyReportUI.tscn").instantiate()
    add_child(report_ui)
    report_ui.display_report(report)
```

### 时间流速控制

```gdscript
# TimeControlSystem.gd
class_name TimeControlSystem extends Node

enum TimeSpeed {
    PAUSED = 0,
    NORMAL = 1,
    FAST_2X = 2,
    FAST_5X = 5,
    ULTRA_10X = 10
}

var current_speed: TimeSpeed = TimeSpeed.NORMAL
var time_scale: float = 1.0

# 改变时间流速
func set_time_speed(speed: TimeSpeed):
    current_speed = speed
    time_scale = float(speed)

    # 更新游戏时间系统
    TimeSystem.set_time_scale(time_scale)

    # 更新UI显示
    update_time_speed_ui()

# 暂停时间
func pause_time():
    set_time_speed(TimeSpeed.PAUSED)

    # 暂停时玩家可以:
    # - 查看AI详细数据
    # - 回放历史事件
    # - 调整相机位置
    # 但AI完全静止

# 快进到特定时间
func fast_forward_to_time(target_hour: int):
    # 以10x速度快进到指定时间
    set_time_speed(TimeSpeed.ULTRA_10X)

    while TimeSystem.current_hour != target_hour:
        await get_tree().create_timer(0.1).timeout

    set_time_speed(TimeSpeed.NORMAL)

# 快进到下一个事件
func fast_forward_to_next_event():
    var next_event = EventSystem.get_next_scheduled_event()

    if next_event:
        fast_forward_to_time(next_event["hour"])
```

## 💬 有限互动系统

### 神秘访客模式

玩家可以以"神秘访客"身份与AI对话,但AI会根据自己的意愿决定是否理会。

```gdscript
# PlayerInteractionSystem.gd
class_name PlayerInteractionSystem extends Node

# 尝试与AI对话
func try_talk_to_ai(ai: Node) -> bool:
    # AI会根据性格、心情、当前状态决定是否接受对话
    var personality = CharacterPersonality.get_personality(ai.name)
    var willingness = calculate_talk_willingness(ai, personality)

    if randf() < willingness:
        start_conversation_with_ai(ai)
        return true
    else:
        show_ai_rejection(ai)
        return false

# 计算AI愿意对话的概率
func calculate_talk_willingness(ai: Node, personality: Dictionary) -> float:
    var base_willingness = personality.get("extraversion", 50) / 100.0

    # 心情影响
    if ai.mood > 70:
        base_willingness += 0.2
    elif ai.mood < 30:
        base_willingness -= 0.3

    # 忙碌状态影响
    if ai.is_working:
        base_willingness -= 0.4
    elif ai.current_activity == "socializing":
        base_willingness += 0.3

    # 如果AI正在与其他AI对话,更不愿意被打断
    if ai.is_in_conversation:
        base_willingness -= 0.5

    return clamp(base_willingness, 0.0, 1.0)

# AI拒绝对话的反应
func show_ai_rejection(ai: Node):
    var rejection_messages = [
        "*{name}看了你一眼,继续忙自己的事*",
        "*{name}礼貌地点了点头,但没有停下来*",
        "*{name}似乎心情不好,不想说话*",
        "*{name}正在工作,无暇理会*"
    ]

    var message = rejection_messages[randi() % rejection_messages.size()]
    message = message.replace("{name}", ai.name)

    UI.show_notification(message)

# 与AI对话
func start_conversation_with_ai(ai: Node):
    # 打开对话UI
    var dialogue_ui = preload("res://scene/ui/PlayerAIDialogue.tscn").instantiate()
    add_child(dialogue_ui)

    # AI会记住与玩家的对话
    MemoryManager.add_memory(ai,
        "今天有个神秘的陌生人和我搭话。",
        MemoryManager.MemoryType.SOCIAL
    )

    # AI的回应基于LLM生成
    dialogue_ui.start_dialogue(ai, "player")
```

### 节日观察模式

在节日期间,玩家观察AI如何参与节日活动。

```gdscript
# FestivalObserver.gd
class_name FestivalObserver extends Node

# 观察节日
func observe_festival(festival_id: String):
    var festival = EventSystem.FESTIVALS[festival_id]

    # 显示节日观察UI
    var festival_ui = preload("res://scene/ui/FestivalObserver.tscn").instantiate()
    add_child(festival_ui)

    # 追踪所有AI在节日中的活动
    track_ai_festival_activities(festival_id)

# 追踪AI节日活动
func track_ai_festival_activities(festival_id: String):
    var all_ai = get_tree().get_nodes_in_group("character")
    var activity_log = []

    for ai in all_ai:
        # 观察AI选择什么活动
        var chosen_activity = ai.choose_festival_activity(festival_id)

        activity_log.append({
            "ai": ai.name,
            "activity": chosen_activity,
            "time": TimeSystem.current_time
        })

        # 如果AI参加舞会,观察谁会邀请谁跳舞
        if chosen_activity == "dance":
            observe_ai_dance_invitation(ai)

# 观察AI舞会邀请
func observe_ai_dance_invitation(ai: Node):
    # AI会邀请关系最好的异性跳舞
    var potential_partners = ai.get_dance_candidates()

    if potential_partners.size() > 0:
        var chosen_partner = ai.choose_dance_partner(potential_partners)

        # 检查对方是否接受
        var accepted = chosen_partner.accept_dance_invitation(ai)

        if accepted:
            log_event("dance_pair", {"ai_a": ai.name, "ai_b": chosen_partner.name})
            # 玩家可以观看AI跳舞
            show_dance_animation(ai, chosen_partner)
        else:
            log_event("dance_rejected", {"requester": ai.name, "rejecter": chosen_partner.name})
```

```gdscript
    # 检测重要事件并通知玩家
    var events_to_watch = [
        "ai_confession",     # AI表白
        "ai_marriage",       # AI结婚
        "ai_breakup",        # AI分手
        "ai_job_change",     # AI换工作
        "ai_conflict",       # AI冲突
        "ai_business_start"  # AI创业
    ]

    # 当重要事件发生时弹出通知
    for event_type in events_to_watch:
        if EventSystem.has_event(event_type):
            show_event_notification(event_type)

func show_event_notification(event_type: String):
    var notification = {
        "title": get_event_title(event_type),
        "message": get_event_message(event_type),
        "action": "前往观看"
    }

    UI.show_notification_popup(notification)

    # 玩家可以点击传送到事件发生地点
```
