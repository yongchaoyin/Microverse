```gdscript
# EventReplaySystem.gd
class_name EventReplaySystem extends Node

var event_history: Array = []

func record_event(event_type: String, data: Dictionary):
    var event = {
        "timestamp": TimeSystem.get_current_timestamp(),
        "type": event_type,
        "data": data,
        "participants": data.get("participants", [])
    }

    event_history.append(event)

func get_events_by_type(type: String) -> Array:
    return event_history.filter(func(e): return e["type"] == type)

func get_events_by_ai(ai_name: String) -> Array:
    return event_history.filter(
        func(e): return ai_name in e["participants"]
    )
```

```gdscript
    var base = personality.get("extraversion", 50) / 100.0

    if ai.mood > 70:

    elif ai.mood < 30:
        base -= 0.3

    # 忙碌状态影响
    if ai.is_working:
        base -= 0.4

    return clamp(base, 0.0, 1.0)
```

## 🔧 第一阶段实现细节

### 1. TimeSystem.gd 实现（带观察者控制）

```gdscript
# script/core/TimeSystem.gd
extends Node

class_name TimeSystem

# ===== 常量 =====
const REAL_SECONDS_PER_GAME_MINUTE = 0.5
const DAYS_PER_SEASON = 28
const SEASONS = ["Spring", "Summer", "Autumn", "Winter"]

# ===== 信号 =====
signal time_changed(hour: int, minute: int)
signal new_day(day: int, season: String, year: int)
signal new_season(season: String, year: int)
signal morning_started()
signal noon_started()
signal evening_started()
signal night_started()

# ===== 时间变量 =====
var current_year: int = 1
var current_season: int = 0
var current_day: int = 1
var current_hour: int = 6
var current_minute: int = 0

var time_speed_multiplier: float = 1.0  # 观察者可调节: 0(暂停), 1, 2, 5, 10
var _time_accumulator: float = 0.0

# ===== 观察者控制功能 =====
func set_time_speed(speed: float):
    """观察者设置时间流速"""
    time_speed_multiplier = clamp(speed, 0.0, 10.0)
    print("[TimeSystem] 时间流速设置为: %sx" % speed)

func pause_time():
    """观察者暂停时间"""
    time_speed_multiplier = 0.0

func resume_time():
    """观察者恢复时间"""
    time_speed_multiplier = 1.0

func fast_forward_to_hour(target_hour: int):
    """观察者快进到指定时间"""
    var original_speed = time_speed_multiplier
    time_speed_multiplier = 10.0  # 10x快进

    while current_hour != target_hour:
        await get_tree().create_timer(0.1).timeout

    time_speed_multiplier = original_speed

# ===== 生命周期 =====
func _ready():
    print("[TimeSystem] 时间系统初始化（观察者模式）")

func _process(delta: float):
    if time_speed_multiplier <= 0:
        return  # 时间暂停

    _time_accumulator += delta * time_speed_multiplier

    while _time_accumulator >= REAL_SECONDS_PER_GAME_MINUTE:
        _time_accumulator -= REAL_SECONDS_PER_GAME_MINUTE
        _advance_minute()

# ===== 时间推进 =====
func _advance_minute():
    current_minute += 1

    if current_minute >= 60:
        current_minute = 0
        _advance_hour()

    time_changed.emit(current_hour, current_minute)

func _advance_hour():
    current_hour += 1

    # 触发时间段信号（AI会响应）
    match current_hour:
        6: morning_started.emit()
        12: noon_started.emit()
        18: evening_started.emit()
        22: night_started.emit()

    if current_hour >= 24:
        current_hour = 0
        _advance_day()

func _advance_day():
    current_day += 1

    if current_day > DAYS_PER_SEASON:
        current_day = 1
        _advance_season()

    new_day.emit(current_day, get_current_season_name(), current_year)

func _advance_season():
    current_season += 1

    if current_season >= 4:
        current_season = 0
        current_year += 1

    new_season.emit(get_current_season_name(), current_year)

# ===== 工具函数 =====
func get_current_season_name() -> String:
    return SEASONS[current_season]

func get_time_string() -> String:
    return "%02d:%02d" % [current_hour, current_minute]

func get_date_string() -> String:
    return "%s %d日，第%d年" % [get_current_season_name(), current_day, current_year]

func get_current_timestamp() -> Dictionary:
    return {
        "year": current_year,
        "season": current_season,
        "day": current_day,
        "hour": current_hour,
        "minute": current_minute
    }
```

## ⚡ 性能优化方案

### AI API调用优化

**问题**: 大量AI角色频繁调用LLM API会导致：
- 成本过高
- 响应延迟
- 超过API速率限制

**解决方案**:

#### 1. 智能缓存系统

```gdscript
# script/ai/AIResponseCache.gd
class_name AIResponseCache extends Node

var cache: Dictionary = {}
var cache_ttl: Dictionary = {}

const CACHE_DURATION = 3600  # 缓存1小时

func get_cached_response(prompt: String) -> String:
    var hash = prompt.md5_text()

    if cache.has(hash):
        if Time.get_unix_time_from_system() < cache_ttl[hash]:
            return cache[hash]
        else:
            cache.erase(hash)
            cache_ttl.erase(hash)

    return ""

func cache_response(prompt: String, response: String):
    var hash = prompt.md5_text()
    cache[hash] = response
    cache_ttl[hash] = Time.get_unix_time_from_system() + CACHE_DURATION
```

#### 2. AI决策降频

```gdscript
# AIAgent.gd

# AI只在必要时调用LLM
func should_call_llm() -> bool:
```
