## ⏰ 时间系统

### 系统概述

时间系统是游戏的心跳，驱动所有循环性内容。

### 时间单位

```gdscript
class_name TimeSystem extends Node

# 时间常量
const MINUTES_PER_DAY = 1440        # 24小时 * 60分钟
const REAL_SECONDS_PER_GAME_MINUTE = 0.5  # 现实1秒 = 游戏2分钟
const DAYS_PER_SEASON = 28
const SEASONS = ["Spring", "Summer", "Autumn", "Winter"]

# 当前时间
var current_year: int = 1
var current_season: int = 0  # 0=春, 1=夏, 2=秋, 3=冬
var current_day: int = 1     # 1-28
var current_hour: int = 6    # 0-23
var current_minute: int = 0  # 0-59

# 时间流逝速度 (可调节)
var time_speed_multiplier: float = 1.0  # 1.0=正常, 2.0=2倍速, 0=暂停
```

### 时间事件

```gdscript
# 信号
signal time_changed(hour: int, minute: int)
signal new_day(day: int, season: String)
signal new_season(season: String, year: int)
signal new_year(year: int)

# 时间段标记
signal morning_started()   # 6:00
signal noon_started()      # 12:00
signal evening_started()   # 18:00
signal night_started()     # 22:00
```

### 时间段定义

| 时间段 | 时间范围 | AI行为模式 | 观察者可做事项 |
|-------|---------|-----------|------------|
| **黎明** | 6:00-8:00 | AI起床、洗漱、吃早餐 | 观察AI晨间活动 |
| **上午** | 8:00-12:00 | AI工作/学习 | 观察工作场所互动 |
| **午间** | 12:00-14:00 | AI午餐、午休 | 观察社交、餐厅活动 |
| **下午** | 14:00-18:00 | AI继续工作 | 观察AI下午活动 |
| **傍晚** | 18:00-22:00 | AI下班、娱乐、社交 | 观察社交网络变化 |
| **深夜** | 22:00-6:00 | AI回家睡觉 | 查看数据报告、时光回放 |

### 实现要点

```gdscript
# TimeSystem.gd 核心逻辑

func _process(delta: float):
    if time_speed_multiplier <= 0:
        return  # 时间暂停

    # 累积真实时间
    var adjusted_delta = delta * time_speed_multiplier
    _time_accumulator += adjusted_delta

    # 每0.5秒推进1游戏分钟（默认）
    while _time_accumulator >= REAL_SECONDS_PER_GAME_MINUTE:
        _time_accumulator -= REAL_SECONDS_PER_GAME_MINUTE
        _advance_minute()

func _advance_minute():
    current_minute += 1

    if current_minute >= 60:
        current_minute = 0
        current_hour += 1
        _check_hour_events()

    if current_hour >= 24:
        current_hour = 0
        current_day += 1
        new_day.emit(current_day, get_current_season_name())
        _trigger_daily_events()

    if current_day > DAYS_PER_SEASON:
        current_day = 1
        current_season += 1

        if current_season >= 4:
            current_season = 0
            current_year += 1
            new_year.emit(current_year)

        new_season.emit(get_current_season_name(), current_year)

    time_changed.emit(current_hour, current_minute)
```

```gdscript
        money_changed.emit(player_money, -amount)
```

```gdscript
# ShopSystem.gd
class_name ShopSystem extends Node

# 商店数据
var shops = {
    "Pierre": {
        "name_cn": "皮埃尔杂货店",
        "owner": "NPC_Pierre",  # 未来可以是AI角色

        }
    },
    "Blacksmith": {
        "name_cn": "铁匠铺",
        "owner": "NPC_Clint",
        "open_hours": {"start": 9, "end": 16},
        "closed_days": ["Friday"],
        "stock": {
            "Copper_Bar": {"quantity": 5, "price_multiplier": 1.2},
            "Iron_Bar": {"quantity": 3, "price_multiplier": 1.5}
        }
    }
}

# 检查商店是否营业
func is_shop_open(shop_id: String) -> bool:
    var shop = shops[shop_id]
    var time_system = get_node("/root/TimeSystem")

    # 检查营业时间
    if time_system.current_hour < shop["open_hours"]["start"]:
        return false
    if time_system.current_hour >= shop["open_hours"]["end"]:
        return false

    # 检查休息日
    var day_name = time_system.get_day_of_week()
    if day_name in shop["closed_days"]:
        return false

    return true

# 购买物品
func buy_item(shop_id: String, item_id: String, quantity: int = 1) -> bool:
    if not is_shop_open(shop_id):
        print("[Shop] %s 当前未营业" % shops[shop_id]["name_cn"])
        return false

    var shop = shops[shop_id]
    var stock_info = shop["stock"].get(item_id)

    if not stock_info:
        print("[Shop] 商店没有出售 %s" % item_id)
        return false

    # 检查库存
    if stock_info["quantity"] != -1 and stock_info["quantity"] < quantity:
        print("[Shop] 库存不足")
        return false

    # 计算价格
    var base_price = ItemData.get_sell_price(item_id)
    var actual_price = int(base_price * stock_info["price_multiplier"]) * quantity

    # 扣款
    var economy = get_node("/root/EconomyManager")
    if economy.remove_money(actual_price, "购买 %s x%d" % [item_id, quantity]):
        # 添加到玩家背包
        var inventory = get_node("/root/InventoryManager")
        inventory.add_item(item_id, quantity)

        # 减少库存
        if stock_info["quantity"] != -1:
            stock_info["quantity"] -= quantity

        return true

    return false
```

### AI决策树扩展

当前AI每60秒做一次决策，需要增强决策的复杂度和多样性。

```gdscript
# AIAgent.gd 扩展

# 决策权重系统
func make_enhanced_decision():
    var decision_factors = evaluate_decision_factors()
    var available_actions = get_available_actions()

    # 计算每个行动的权重
    var weighted_actions = []
    for action in available_actions:
        var weight = calculate_action_weight(action, decision_factors)
        weighted_actions.append({"action": action, "weight": weight})

    # 按权重排序
    weighted_actions.sort_custom(func(a, b): return a.weight > b.weight)

    # 加入随机性（AI不应该完全可预测）
    var chosen_action = weighted_random_choice(weighted_actions)
    execute_action(chosen_action)

# 评估决策因素
func evaluate_decision_factors() -> Dictionary:
    return {
        "energy": character_data["stats"]["energy"],      # 精力值
        "mood": character_data["stats"]["mood"],          # 心情
        "hunger": character_data["stats"]["hunger"],      # 饥饿度
        "social_need": character_data["stats"]["social_need"],  # 社交需求
        "work_pressure": character_data["career"]["pressure"],  # 工作压力

        "relationships": get_top_relationships(),          # 关系状态
    }

# 可用行动列表
func get_available_actions() -> Array:
    var actions = []
    var time_period = get_time_period()

    # 基础行动（总是可用）
    actions.append_array([
        {"type": "idle", "desc": "闲逛"},
        {"type": "move_random", "desc": "随机移动"}
    ])

    # 时间相关行动
    if time_period in ["morning", "afternoon"]:
        actions.append({"type": "work", "desc": "工作"})

    if time_period == "evening":
        actions.append({"type": "socialize", "desc": "社交"})
        actions.append({"type": "go_to_bar", "desc": "去酒馆"})

    # 需求驱动行动
    if character_data["stats"]["hunger"] > 70:
        actions.append({"type": "eat", "desc": "吃饭"})

    if character_data["stats"]["energy"] < 30:
        actions.append({"type": "rest", "desc": "休息"})

    # 社交相关
    var nearby_chars = get_nearby_characters()
    if nearby_chars.size() > 0:
        for char in nearby_chars:
            var rel = get_relationship_with(char.name)
            if rel and rel["friendship"] > 200:
                actions.append({"type": "chat", "target": char, "desc": "与%s聊天" % char.name})

    # 任务相关
    if character_data["tasks"].size() > 0:
        actions.append({"type": "do_task", "desc": "完成任务"})

    return actions

# 计算行动权重
func calculate_action_weight(action: Dictionary, factors: Dictionary) -> float:
    var base_weight = 10.0

    match action["type"]:
        "work":
            # 工作时间段权重高
            if factors["time_of_day"] in ["morning", "afternoon"]:
                base_weight += 50
            # 工作压力大时权重降低
            base_weight -= factors["work_pressure"] * 0.3
            # 精力低时不想工作
            if factors["energy"] < 40:
                base_weight -= 30

        "eat":
            # 饥饿度越高权重越大
            base_weight += factors["hunger"]
            if factors["hunger"] > 80:
                base_weight += 50  # 非常饥饿时优先吃饭

        "socialize", "chat":
            # 社交需求高时权重大
            base_weight += factors["social_need"]
            # 傍晚时段社交权重高
            if factors["time_of_day"] == "evening":
                base_weight += 30
            # 心情好时更愿意社交
            base_weight += factors["mood"] * 0.5

        "rest":
            # 精力低时权重高
            base_weight += (100 - factors["energy"])
            # 深夜时权重极高
            if factors["time_of_day"] == "night":
                base_weight += 80

        "do_task":
            # 任务越多权重越高
            base_weight += factors["pending_tasks"].size() * 10
            # 责任心强的角色更愿意完成任务（从personality读取）
            var personality = CharacterPersonality.get_personality(character.name)
            base_weight += personality.get("responsibility", 50)

    return max(base_weight, 0)  # 权重不能为负
```

```gdscript
# InventoryManager.gd
class_name InventoryManager extends Node

const MAX_SLOTS = 36  # 背包格子数（可升级）

var inventory_items = []  # 物品列表

# 物品槽数据结构
class InventorySlot:
    var item_id: String = ""
    var quantity: int = 0
    var quality: ItemData.ItemQuality = ItemData.ItemQuality.NORMAL
    var metadata: Dictionary = {}  # 额外信息（如工具耐久度）

# 初始化背包
func _ready():
    for i in range(MAX_SLOTS):
        inventory_items.append(InventorySlot.new())

# 添加物品
func add_item(item_id: String, quantity: int = 1, quality: ItemData.ItemQuality = ItemData.ItemQuality.NORMAL) -> bool:
    # 1. 尝试堆叠到现有物品
    for slot in inventory_items:
        if slot.item_id == item_id and slot.quality == quality:
            var item_data = ItemData.ITEMS_DATABASE[item_id]
            var max_stack = item_data.get("max_stack", 999)

            if slot.quantity < max_stack:
                var can_add = min(quantity, max_stack - slot.quantity)
                slot.quantity += can_add
                quantity -= can_add

                if quantity <= 0:
                    emit_signal("item_added", item_id, can_add)
                    return true

    # 2. 放入空格子
    while quantity > 0:
        var empty_slot = find_empty_slot()
        if empty_slot == null:
            print("[Inventory] 背包已满")
            return false

        var item_data = ItemData.ITEMS_DATABASE[item_id]
        var max_stack = item_data.get("max_stack", 999)
        var add_amount = min(quantity, max_stack)

        empty_slot.item_id = item_id
        empty_slot.quantity = add_amount
        empty_slot.quality = quality
        quantity -= add_amount

    emit_signal("item_added", item_id, quantity)
    return true

# 移除物品
func remove_item(item_id: String, quantity: int = 1) -> bool:
    var remaining = quantity

    for slot in inventory_items:
        if slot.item_id == item_id:
            if slot.quantity >= remaining:
                slot.quantity -= remaining
                if slot.quantity == 0:
                    slot.item_id = ""
                    slot.quality = ItemData.ItemQuality.NORMAL
                emit_signal("item_removed", item_id, quantity)
                return true
            else:
                remaining -= slot.quantity
                slot.quantity = 0
                slot.item_id = ""

    return false  # 数量不足

# 查找空格子
func find_empty_slot() -> InventorySlot:
    for slot in inventory_items:
        if slot.item_id == "":
            return slot
    return null
```
